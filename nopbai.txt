package com.example.th6;

import android.os.Bundle;
import android.content.Intent;
import android.text.TextUtils;

import androidx.appcompat.app.AppCompatActivity;
import androidx.fragment.app.Fragment;

import com.example.th6.data.AuthRepository;
import com.example.th6.databinding.ActivityMainBinding;
import com.example.th6.fragments.AlbumFragment;
import com.example.th6.fragments.FavoritesFragment;
import com.example.th6.fragments.LibraryFragment;
import com.example.th6.fragments.ProfileFragment;
import com.example.th6.fragments.TrashFragment;
import com.example.th6.activities.LoginActivity;
import com.example.th6.services.BackupWorkScheduler;
import com.example.th6.services.MediaStoreBackupObserver;
import com.example.th6.utils.BackupPreferences;

public class MainActivity extends AppCompatActivity {

    private ActivityMainBinding binding;
    private boolean sessionVerified;
    private boolean activityStarted;
    private String verifiedUserId;
    private MediaStoreBackupObserver mediaStoreBackupObserver;

    @Override
    protected void onStart() {
        super.onStart();
        activityStarted = true;
        startMediaStoreObserverIfEligible();
    }

    @Override
    protected void onStop() {
        activityStarted = false;
        stopMediaStoreObserver();
        super.onStop();
    }

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        binding = ActivityMainBinding.inflate(getLayoutInflater());
        setContentView(binding.getRoot());

        AuthRepository authRepository = new AuthRepository();
        binding.bottomNav.setEnabled(false);

        binding.bottomNav.setOnItemSelectedListener(item -> {
            if (!sessionVerified) return false;
            int itemId = item.getItemId();
            if (itemId == R.id.nav_library) {
                loadFragment(new LibraryFragment());
                return true;
            } else if (itemId == R.id.nav_albums) {
                loadFragment(new AlbumFragment());
                return true;
            } else if (itemId == R.id.nav_trash) {
                loadFragment(new TrashFragment());
                return true;
            } else if (itemId == R.id.nav_favorites) {
                loadFragment(new FavoritesFragment());
                return true;
            } else if (itemId == R.id.nav_profile) {
                loadFragment(new ProfileFragment());
                return true;
            }
            return false;
        });

        authRepository.checkIsLoggedInAsync(new AuthRepository.SessionCallback() {
            @Override
            public void onResult(boolean isLoggedIn, String userId, String errorMessage) {
                if (isFinishing() || isDestroyed()) return;
                if (!isLoggedIn || TextUtils.isEmpty(userId)) {
                    Intent intent = new Intent(MainActivity.this, LoginActivity.class);
                    intent.setFlags(Intent.FLAG_ACTIVITY_NEW_TASK | Intent.FLAG_ACTIVITY_CLEAR_TASK);
                    if (!TextUtils.isEmpty(errorMessage)) intent.putExtra("auth_session_error", errorMessage);
                    startActivity(intent);
                    finish();
                    return;
                }

                sessionVerified = true;
                verifiedUserId = userId;
                binding.bottomNav.setEnabled(true);
                BackupWorkScheduler.reconcile(MainActivity.this, userId, true);
                startMediaStoreObserverIfEligible();
                if (getSupportFragmentManager().findFragmentById(R.id.fragmentContainer) == null) {
                    loadFragment(new LibraryFragment());
                }
            }
        });
    }

    public void selectTab(int navItemId) {
        if (binding != null && binding.bottomNav != null) {
            binding.bottomNav.setSelectedItemId(navItemId);
        }
    }

    private void loadFragment(Fragment fragment) {
        getSupportFragmentManager()
                .beginTransaction()
                .replace(R.id.fragmentContainer, fragment)
                .commit();
    }

    private void startMediaStoreObserverIfEligible() {
        if (!activityStarted || !sessionVerified || verifiedUserId == null || verifiedUserId.isEmpty()) return;
        BackupPreferences preferences = new BackupPreferences(this, verifiedUserId);
        if (!preferences.isBackupEnabled() || preferences.isPaused()) return;
        if (mediaStoreBackupObserver != null) return;
        mediaStoreBackupObserver = new MediaStoreBackupObserver(getApplicationContext(), verifiedUserId);
        mediaStoreBackupObserver.register();
    }

    private void stopMediaStoreObserver() {
        if (mediaStoreBackupObserver != null) {
            mediaStoreBackupObserver.unregister();
            mediaStoreBackupObserver = null;
        }
    }

    @Override
    protected void onDestroy() {
        stopMediaStoreObserver();
        super.onDestroy();
    }
}

    private void observeBackupWork() {
        UserInfo user = authRepository.getCurrentUser();
        if (user == null) {
            renderBackupStatus();
            return;
        }
        WorkManager workManager = WorkManager.getInstance(requireContext().getApplicationContext());
        String uid = user.getId();
        workManager.getWorkInfosForUniqueWorkLiveData(BackupWorkScheduler.immediateWorkName(uid))
                .observe(getViewLifecycleOwner(), infos -> {
                    immediateBackupWork = latestWorkInfo(infos);
                    renderBackupStatus();
                });
        workManager.getWorkInfosForUniqueWorkLiveData(BackupWorkScheduler.periodicWorkName(uid))
                .observe(getViewLifecycleOwner(), infos -> {
                    periodicBackupWork = latestWorkInfo(infos);
                    renderBackupStatus();
                });
    }

    @Nullable
    private WorkInfo latestWorkInfo(@Nullable List<WorkInfo> infos) {
        if (infos == null || infos.isEmpty()) return null;
        for (int i = infos.size() - 1; i >= 0; i--) {
            if (infos.get(i).getState() == WorkInfo.State.RUNNING) return infos.get(i);
        }
        return infos.get(infos.size() - 1);
    }

    private void renderBackupStatus() {
        if (binding == null) return;
        WorkInfo running = isRunning(immediateBackupWork) ? immediateBackupWork
                : isRunning(periodicBackupWork) ? periodicBackupWork : null;
        if (running != null) {
            binding.layoutBackupStatus.setVisibility(View.VISIBLE);
            binding.progressBackupStatus.setVisibility(View.VISIBLE);
            binding.tvBackupStatus.setText("Đang sao lưu ảnh");
            int total = running.getProgress().getInt("total", 0);
            int completed = running.getProgress().getInt("completed", 0);
            boolean videoUpload = "video".equals(running.getProgress().getString("current_type"));
            binding.tvBackupStatus.setText(videoUpload ? "Đang sao lưu video" : "Đang sao lưu media");
            if (total > 0) {
                binding.tvBackupProgress.setVisibility(View.VISIBLE);
                binding.tvBackupProgress.setText("Đã xử lý " + Math.min(completed, total) + "/" + total + " media");
            } else {
                binding.tvBackupProgress.setVisibility(View.GONE);
            }
            return;
        }
        binding.layoutBackupStatus.setVisibility(View.GONE);
        binding.progressBackupStatus.setVisibility(View.GONE);
        maybeReportPartialCompletion(immediateBackupWork, false);
        maybeReportPartialCompletion(periodicBackupWork, true);
    }

    private boolean isRunning(@Nullable WorkInfo info) {
        return info != null && info.getState() == WorkInfo.State.RUNNING;
    }

    private void maybeReportPartialCompletion(@Nullable WorkInfo info, boolean periodic) {
        if (info == null || binding == null) return;
        boolean completed = info.getState() == WorkInfo.State.SUCCEEDED
                || (periodic && info.getState() == WorkInfo.State.ENQUEUED);
        if (!completed) return;
        String id = info.getId().toString();
        int failures = info.getOutputData().getInt("failures", 0);
        if (failures <= 0) return;
        UserInfo user = authRepository.getCurrentUser();
        if (user == null) return;
        android.content.SharedPreferences noticePrefs = requireContext().getSharedPreferences(
                "backup_status_notices", android.content.Context.MODE_PRIVATE);
        String noticeKey = "last_partial_" + user.getId();
        if (id.equals(noticePrefs.getString(noticeKey, null))) return;
        noticePrefs.edit().putString(noticeKey, id).apply();
        Snackbar.make(binding.getRoot(), "Sao lưu hoàn tất một phần · " + failures + " ảnh cần xử lý", Snackbar.LENGTH_LONG)
                .setAction("Chi tiết", v -> {
                    if (isAdded()) startActivity(new Intent(requireContext(), BackupCenterActivity.class));
                }).show();
    }

    private void checkPermissionAndScan() {
        if (!shouldRequestMediaPermissions()) {
            scanDeviceGallery();
        } else {
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.UPSIDE_DOWN_CAKE) {
                requestPermissionLauncher.launch(new String[]{Manifest.permission.READ_MEDIA_IMAGES,
                        Manifest.permission.READ_MEDIA_VIDEO, Manifest.permission.READ_MEDIA_VISUAL_USER_SELECTED});
            } else if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
                requestPermissionLauncher.launch(new String[]{Manifest.permission.READ_MEDIA_IMAGES,
                        Manifest.permission.READ_MEDIA_VIDEO});
            } else {
                requestPermissionLauncher.launch(new String[]{Manifest.permission.READ_EXTERNAL_STORAGE});
            }
        }
    }

    private boolean shouldRequestMediaPermissions() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.UPSIDE_DOWN_CAKE) {
            boolean fullImageAccess = hasPermission(Manifest.permission.READ_MEDIA_IMAGES);
            boolean fullVideoAccess = hasPermission(Manifest.permission.READ_MEDIA_VIDEO);
            boolean selectedAccess = hasPermission(Manifest.permission.READ_MEDIA_VISUAL_USER_SELECTED);
            return !selectedAccess && (!fullImageAccess || !fullVideoAccess);
        }
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
            return !hasPermission(Manifest.permission.READ_MEDIA_IMAGES)
                    || !hasPermission(Manifest.permission.READ_MEDIA_VIDEO);
        }
        return !hasPermission(Manifest.permission.READ_EXTERNAL_STORAGE);
    }

    private boolean hasAnyMediaReadPermission() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.UPSIDE_DOWN_CAKE) {
            return hasPermission(Manifest.permission.READ_MEDIA_IMAGES)
                    || hasPermission(Manifest.permission.READ_MEDIA_VIDEO)
                    || hasPermission(Manifest.permission.READ_MEDIA_VISUAL_USER_SELECTED);
        }
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
            return hasPermission(Manifest.permission.READ_MEDIA_IMAGES)
                    || hasPermission(Manifest.permission.READ_MEDIA_VIDEO);
        }
        return hasPermission(Manifest.permission.READ_EXTERNAL_STORAGE);
    }

    private boolean hasPermission(String permission) {
        return ContextCompat.checkSelfPermission(requireContext(), permission) == PackageManager.PERMISSION_GRANTED;
    }

    private void scanDeviceGallery() {
        if (binding == null || scanInProgress) return;
        scanInProgress = true;
        binding.progressBar.setVisibility(View.VISIBLE);
        photoScanner.scanPhotos(requireContext(), new DevicePhotoScanner.ScanCallback() {
            @Override
            public void onScanComplete(List<LocalMediaPhoto> localPhotos) {
                scanInProgress = false;
                if (binding == null || !isAdded()) return;
                binding.progressBar.setVisibility(View.GONE);
                binding.swipeRefresh.setRefreshing(false);

                localDevicePhotos = localPhotos != null ? new ArrayList<>(localPhotos) : new ArrayList<>();
                UserInfo user = authRepository.getCurrentUser();
                allDevicePhotos = user == null
                        ? filterLocalsInTrash(requireContext(), localDevicePhotos, knownTrash(null))
                        : filterLocalsInTrash(requireContext(), localDevicePhotos, knownTrash(user.getId()));
                applyViewModeAndSort();
                updateEmptyState();
                loadCloudPhotosForCurrentUser();
                if (user != null) {
                    // Enqueue only after a successful device scan. WorkManager runs the
                    // existing worker off the UI thread and rechecks the backup queue.
                    BackupWorkScheduler.enqueueIfEligible(requireContext(), user.getId());
                }
            }

            @Override
            public void onScanError(String error) {
                scanInProgress = false;
                if (binding == null || !isAdded()) return;
                binding.progressBar.setVisibility(View.GONE);
                binding.swipeRefresh.setRefreshing(false);
                // A failed scan cannot validate the previous MediaStore snapshot. Drop those
                // local rows so removed files are not presented as if they still existed;
                // the independent cloud query below can still contribute verified cloud rows.
                localDevicePhotos = new ArrayList<>();
                UserInfo user = authRepository.getCurrentUser();
                allDevicePhotos = filterLocalsInTrash(requireContext(), localDevicePhotos,
                        knownTrash(user == null ? null : user.getId()));
                applyViewModeAndSort();
                updateEmptyState();
                loadCloudPhotosForCurrentUser();
                Toast.makeText(getContext(), error, Toast.LENGTH_SHORT).show();
            }
        });
    }

    private void uploadPhotoToSupabaseStorage(Uri imageUri) {
        if (manualUploadInProgress || binding == null || !isAdded()) return;
        UserInfo currentUser = authRepository.isLoggedIn() ? authRepository.getCurrentUser() : null;
        if (currentUser == null) {
            Toast.makeText(getContext(), "Hãy đăng nhập để tải ảnh lên cloud.", Toast.LENGTH_LONG).show();
            return;
        }
        manualUploadInProgress = true;
        binding.progressBar.setVisibility(View.VISIBLE);
        Context appContext = requireContext().getApplicationContext();
        String mime = appContext.getContentResolver().getType(imageUri);
        if (mime != null && !mime.startsWith("image/")) {
            manualUploadInProgress = false;
            binding.progressBar.setVisibility(View.GONE);
            Toast.makeText(getContext(), "Chỉ có thể tải ảnh lên từ thao tác này.", Toast.LENGTH_LONG).show();
            return;
        }
        if (mime == null) mime = "image/jpeg";
        String extension = android.webkit.MimeTypeMap.getSingleton().getExtensionFromMimeType(mime);
        if (extension == null || extension.isEmpty()) extension = "img";
        String identity = currentUser.getId() + "|" + imageUri;
        String stableId = UUID.nameUUIDFromBytes(identity.getBytes(java.nio.charset.StandardCharsets.UTF_8)).toString();
        String objectPath = currentUser.getId() + "/photos/manual_" + stableId + "." + extension;
        storageRepository.uploadMediaAsync(appContext, imageUri, objectPath, mime, new StorageRepository.UploadCallback() {
            @Override public void onSuccess(String publicUrl) {
                try (BackupUploadLedger ledger = new BackupUploadLedger(appContext)) {
                    ledger.recordUpload(currentUser.getId(), imageUri.toString(), 0L, publicUrl);
                } catch (RuntimeException ignored) { }
                UserInfo activeUser = authRepository.isLoggedIn() ? authRepository.getCurrentUser() : null;
                if (activeUser == null || !currentUser.getId().equals(activeUser.getId())) {
                    manualUploadInProgress = false;
                    if (binding != null) binding.progressBar.setVisibility(View.GONE);
                    return;
                }
                Photo photo = new Photo();
                photo.setId("manual_" + stableId);
                photo.setOwnerUid(currentUser.getId());
                photo.setName("Photo_" + System.currentTimeMillis());
                photo.setImageUrl(publicUrl);
                photo.setDisplayUrl(publicUrl);
                photo.setCreatedAt(System.currentTimeMillis());
                photo.setMediaType("image");
                photo.setLocalMediaUri(imageUri.toString());
                photo.setStorageProvider("supabase");
                photo.setStoragePath(objectPath);
                repository.savePhotoAsync(photo, new PhotoRepository.Callback<Void>() {
                    @Override public void onSuccess(Void result) {
                        try (BackupUploadLedger ledger = new BackupUploadLedger(appContext)) {
                            ledger.remove(currentUser.getId(), imageUri.toString());
                        } catch (RuntimeException ignored) { }
                        manualUploadInProgress = false;
                        if (binding == null || !isAdded()) return;
                        binding.progressBar.setVisibility(View.GONE);
                        Toast.makeText(getContext(), "Đã tải ảnh lên cloud.", Toast.LENGTH_SHORT).show();
                        scanDeviceGallery();
                    }
                    @Override public void onError(String error) {
                        manualUploadInProgress = false;
                        if (binding == null || !isAdded()) return;
                        binding.progressBar.setVisibility(View.GONE);
                        Toast.makeText(getContext(), "Ảnh đã tải lên Storage nhưng chưa lưu metadata. Hãy chọn lại đúng ảnh này để thử đồng bộ lại.", Toast.LENGTH_LONG).show();
                    }
                });
            }
            @Override public void onError(String error) {
                manualUploadInProgress = false;
                if (binding == null || !isAdded()) return;
                binding.progressBar.setVisibility(View.GONE);
                Toast.makeText(getContext(), error, Toast.LENGTH_LONG).show();
            }
        });
    }
    private void showSortPopupMenu(View anchorView) {
        PopupMenu popupMenu = new PopupMenu(requireContext(), anchorView);
        popupMenu.getMenuInflater().inflate(R.menu.options_menu_library, popupMenu.getMenu());

        popupMenu.setOnMenuItemClickListener(item -> {
            int itemId = item.getItemId();
            if (itemId == R.id.sort_date_desc || itemId == R.id.sort_date_asc
                    || itemId == R.id.sort_name_asc || itemId == R.id.sort_name_desc) {
                currentSortMode = itemId;
                requireContext().getSharedPreferences("photovault_library", Context.MODE_PRIVATE)
                        .edit().putInt("sort_mode", currentSortMode).apply();
                applyViewModeAndSort();
                return true;
            } else if (itemId == R.id.action_upload_photo) {
                if (!manualUploadInProgress) photoPickerLauncher.launch("image/*");
                return true;
            } else if (itemId == R.id.action_slideshow) {
                startSlideshow();
                return true;
            } else if (itemId == R.id.action_backup) {
                startActivity(new Intent(requireContext(), BackupCenterActivity.class));
                return true;
            } else if (itemId == R.id.action_smart_cleanup) {
                startActivity(new Intent(requireContext(), SmartCleanupActivity.class));
                return true;
            } else if (itemId == R.id.action_search) {
                startActivity(new Intent(requireContext(), SearchActivity.class));
                return true;
            } else if (itemId == R.id.action_favorites) {
                if (getActivity() instanceof MainActivity) {
                    ((MainActivity) getActivity()).selectTab(R.id.nav_favorites);
                }
                return true;
            } else if (itemId == R.id.action_profile) {
                if (getActivity() instanceof MainActivity) {
                    ((MainActivity) getActivity()).selectTab(R.id.nav_profile);
                }
                return true;
            } else if (itemId == R.id.action_logout) {
                authRepository.signOutAsync(result -> {
                    if (getActivity() == null) return;
                    if (!result.getLocalSessionCleared()) {
                        Toast.makeText(getContext(), "Không thể xóa phiên đăng nhập trên thiết bị. Thử lại.", Toast.LENGTH_LONG).show();
                        return;
                    }
                    if (!result.getServerSignOutConfirmed()) {
                        Toast.makeText(getContext(), "Đã đăng xuất khỏi thiết bị. Máy chủ chưa xác nhận thu hồi phiên.", Toast.LENGTH_LONG).show();
                    }
                    Intent intent = new Intent(getActivity(), LoginActivity.class);
                    intent.setFlags(Intent.FLAG_ACTIVITY_NEW_TASK | Intent.FLAG_ACTIVITY_CLEAR_TASK);
                    startActivity(intent);
                    getActivity().finish();
                });
                return true;
            }
            return false;
        });

        popupMenu.show();
    }

    private void startSlideshow() {
        if (allDevicePhotos.isEmpty()) {
            Toast.makeText(getContext(), "Không có ảnh nào để trình chiếu", Toast.LENGTH_SHORT).show();
            return;
        }

        List<Photo> slideshowList = new ArrayList<>();
        for (LocalMediaPhoto local : allDevicePhotos) {
            Photo p = new Photo();
            p.setId(String.valueOf(local.getMediaId()));
            p.setOwnerUid("local");
            p.setName(local.getDisplayName());
            p.setImageUrl(local.getContentUri());
            p.setDisplayUrl(local.getContentUri());
            p.setCreatedAt(local.getTimelineTimestamp());
            slideshowList.add(p);
        }

        Intent intent = new Intent(requireContext(), SlideshowActivity.class);
        intent.putExtra("photos", (Serializable) slideshowList);
        startActivity(intent);
    }

    private void applyViewModeAndSort() {
        if (allDevicePhotos == null) allDevicePhotos = new ArrayList<>();
        List<LocalMediaPhoto> displayPhotos = MediaTimelineUtils.filterAndSort(allDevicePhotos,
                currentMediaFilter, currentViewMode, currentSortMode,
                LocalMediaPhoto::getTimelineTimestamp, LocalMediaPhoto::getDisplayName, LocalMediaPhoto::isVideo);
        localPhotoAdapter.setPhotoList(displayPhotos, currentViewMode);
        Set<String> validKeys = new LinkedHashSet<>();
        for (LocalMediaPhoto photo : allDevicePhotos) validKeys.add(LocalPhotoAdapter.keyFor(photo));
        selectedPhotoKeys.retainAll(validKeys);
        localPhotoAdapter.setSelectedKeys(selectedPhotoKeys);
        renderSelectionMode();
    }

    private void loadCloudPhotosForCurrentUser() {
        UserInfo user = authRepository.getCurrentUser();
        if (user == null || getContext() == null) {
            displayedCloudUid = null;
            cloudLoadFailed = false;
            if (selectionMode) clearSelection();
            allDevicePhotos = filterLocalsInTrash(requireContext(), localDevicePhotos, knownTrash(null));
            applyViewModeAndSort();
            updateEmptyState();
            return;
        }
        final String requestedUid = user.getId();
        if (!requestedUid.equals(displayedCloudUid)) {
            if (selectionMode) clearSelection();
            allDevicePhotos = filterLocalsInTrash(requireContext(), localDevicePhotos, knownTrash(requestedUid));
            applyViewModeAndSort();
            updateEmptyState();
        }
        repository.getPhotosByUserAsync(requestedUid, new PhotoRepository.Callback<List<Photo>>() {
            @Override public void onSuccess(List<Photo> cloudPhotos) {
                androidx.fragment.app.FragmentActivity activity = getActivity();
                if (activity == null) return;
                cloudLoadFailed = false;
                List<LocalMediaPhoto> localSnapshot = new ArrayList<>(localDevicePhotos);
                List<Photo> cloudSnapshot = cloudPhotos == null ? new ArrayList<>() : new ArrayList<>(cloudPhotos);
                android.content.Context appContext = requireContext().getApplicationContext();
                PhotoVaultTrashCache.replace(appContext, requestedUid, cloudSnapshot);
                cloudMergeExecutor.execute(() -> {
                    List<LocalMediaPhoto> merged = mergeLocalAndCloud(appContext, localSnapshot, cloudSnapshot);
                    activity.runOnUiThread(() -> {
                        UserInfo active = authRepository.getCurrentUser();
                        if (binding == null || !isAdded() || active == null || !requestedUid.equals(active.getId())) return;
                        allDevicePhotos = merged;
                        displayedCloudUid = requestedUid;
                        applyViewModeAndSort();
                        updateEmptyState();
                    });
                });
            }
            @Override public void onError(String error) {
                UserInfo active = authRepository.getCurrentUser();
                if (active == null || !requestedUid.equals(active.getId())) return;
                cloudLoadFailed = true;
                if (binding != null && isAdded()) {
                    updateEmptyState();
                    if (!allDevicePhotos.isEmpty()) {
                        Snackbar.make(binding.getRoot(), "Ảnh trên thiết bị vẫn hiển thị; chưa tải được thư viện cloud.", Snackbar.LENGTH_LONG)
                                .setAction("Thử lại", v -> loadCloudPhotosForCurrentUser()).show();
                    }
                }
            }
        });
    }

    private List<LocalMediaPhoto> mergeLocalAndCloud(android.content.Context context,
                                                       List<LocalMediaPhoto> localItems,
                                                       List<Photo> cloudItems) {
        Map<String, LocalMediaPhoto> byUri = new HashMap<>();
        Map<String, LocalMediaPhoto> byMediaId = new HashMap<>();
        List<LocalMediaPhoto> merged = new ArrayList<>();
        for (LocalMediaPhoto source : localItems) {
            LocalMediaPhoto local = copyLocalPhoto(source);
            merged.add(local);
            if (local.getContentUri() != null) byUri.put(local.getContentUri(), local);
            if (local.getMediaId() > 0) byMediaId.put(localTypeKey(local) + ":" + local.getMediaId(), local);
        }
        Map<String, LocalMediaPhoto> byHash = new HashMap<>();
        Map<String, String> hashByUri = new HashMap<>();
        boolean cloudHasHashes = false;
        for (Photo cloud : cloudItems) if (cloud.getContentHash() != null && !cloud.getContentHash().isEmpty()) cloudHasHashes = true;
        if (cloudHasHashes) {
            for (LocalMediaPhoto local : merged) {
                String hash = hashUri(context, local.getContentUri());
                if (hash != null) {
                    byHash.put(hash, local);
                    hashByUri.put(local.getContentUri(), hash);
                }
            }
        }
        for (Photo cloud : cloudItems) {
            if (cloud.isTrashed() || cloud.getImageUrl() == null || cloud.getImageUrl().isEmpty()) continue;
            LocalMediaPhoto match = cloud.getLocalMediaUri() == null ? null : byUri.get(cloud.getLocalMediaUri());
            String cloudType = cloudType(cloud);
            if (match != null && !cloudType.equals(localTypeKey(match))) match = null;
            if (match == null && cloud.getLocalMediaId() > 0) {
                match = byMediaId.get(cloudType + ":" + cloud.getLocalMediaId());
            }
            if (match == null && cloud.getContentHash() != null) {
                LocalMediaPhoto hashMatch = byHash.get(cloud.getContentHash());
                if (hashMatch != null && cloudType.equals(localTypeKey(hashMatch))) match = hashMatch;
            }
            if (match != null) {
                match.setCloudPhoto(cloud);
                match.setCloudPhotoId(cloud.getId());
                match.setCloudUrl(cloud.getImageUrl());
                match.setBackupStatus(LocalMediaPhoto.STATUS_UPLOADED);
            } else {
                LocalMediaPhoto remote = new LocalMediaPhoto();
                remote.setMediaId(-Math.abs((long) cloud.getId().hashCode()));
                remote.setDisplayName(cloud.getName() == null || cloud.getName().isEmpty() ? "Ảnh cloud" : cloud.getName());
                boolean cloudVideo = "video".equalsIgnoreCase(cloud.getMediaType());
                remote.setVideo(cloudVideo);
                remote.setMimeType(cloudVideo ? "video/*" : "image/*");
                remote.setDateModified(cloud.getCreatedAt());
                remote.setDateTaken(cloud.getCreatedAt());
                remote.setCloudPhoto(cloud);
                remote.setCloudPhotoId(cloud.getId());
                remote.setCloudUrl(cloud.getImageUrl());
                remote.setBackupStatus(LocalMediaPhoto.STATUS_UPLOADED);
                merged.add(remote);
            }
        }
        List<Photo> trashedCloud = new ArrayList<>();
        for (Photo cloud : cloudItems) if (cloud != null && cloud.isTrashed()) trashedCloud.add(cloud);
        trashedCloud.addAll(PhotoVaultTrashCache.getLocal(context));
        removeLocalsMatchedByTrash(merged, trashedCloud, hashByUri);
        return merged;
    }

    private List<Photo> knownTrash(@Nullable String uid) {
        List<Photo> result = PhotoVaultTrashCache.getLocal(requireContext());
        if (uid != null) result.addAll(PhotoVaultTrashCache.get(requireContext(), uid));
        return result;
    }

    private List<LocalMediaPhoto> filterLocalsInTrash(android.content.Context context,
                                                       List<LocalMediaPhoto> locals,
                                                       List<Photo> trashed) {
        List<LocalMediaPhoto> filtered = new ArrayList<>();
        if (locals == null || locals.isEmpty() || trashed == null || trashed.isEmpty()) {
            if (locals != null) filtered.addAll(locals);
            return filtered;
        }
        Map<String, String> hashes = new HashMap<>();
        boolean needsHash = false;
        for (Photo photo : trashed) {
            if (photo.getContentHash() != null && !photo.getContentHash().isEmpty()) needsHash = true;
        }
        if (needsHash) for (LocalMediaPhoto local : locals) {
            String hash = hashUri(context, local.getContentUri());
            if (hash != null) hashes.put(local.getContentUri(), hash);
        }
        Set<String> hiddenUris = new LinkedHashSet<>();
        Set<String> hiddenIds = new java.util.HashSet<>();
        Set<String> hiddenHashes = new java.util.HashSet<>();
        for (Photo photo : trashed) {
            if (photo.getLocalMediaUri() != null && !photo.getLocalMediaUri().isEmpty()) hiddenUris.add(photo.getLocalMediaUri());
            if (photo.getLocalMediaId() > 0) hiddenIds.add(cloudType(photo) + ":" + photo.getLocalMediaId());
            if (photo.getContentHash() != null && !photo.getContentHash().isEmpty()) {
                hiddenHashes.add(cloudType(photo) + ":" + photo.getContentHash());
            }
        }
        for (LocalMediaPhoto local : locals) {
            boolean hidden = (local.getContentUri() != null && hiddenUris.contains(local.getContentUri()))
                    || (local.getMediaId() > 0 && hiddenIds.contains(localTypeKey(local) + ":" + local.getMediaId()))
                    || (local.getContentUri() != null && hiddenHashes.contains(
                    localTypeKey(local) + ":" + hashes.get(local.getContentUri())));
            if (!hidden) filtered.add(local);
        }
        return filtered;
    }

    private void removeLocalsMatchedByTrash(List<LocalMediaPhoto> merged, List<Photo> trashed,
                                             Map<String, String> hashByUri) {
        if (trashed.isEmpty()) return;
        Set<String> hiddenUris = new LinkedHashSet<>();
        Set<String> hiddenIds = new java.util.HashSet<>();
        Set<String> hiddenHashes = new java.util.HashSet<>();
        for (Photo photo : trashed) {
            if (photo.getLocalMediaUri() != null && !photo.getLocalMediaUri().isEmpty()) hiddenUris.add(photo.getLocalMediaUri());
            if (photo.getLocalMediaId() > 0) hiddenIds.add(cloudType(photo) + ":" + photo.getLocalMediaId());
            if (photo.getContentHash() != null && !photo.getContentHash().isEmpty()) {
                hiddenHashes.add(cloudType(photo) + ":" + photo.getContentHash());
            }
        }
        java.util.Iterator<LocalMediaPhoto> iterator = merged.iterator();
        while (iterator.hasNext()) {
            LocalMediaPhoto item = iterator.next();
            if (item.getCloudPhoto() != null) continue;
            String uri = item.getContentUri();
            if ((uri != null && hiddenUris.contains(uri))
                    || (item.getMediaId() > 0 && hiddenIds.contains(localTypeKey(item) + ":" + item.getMediaId()))
                    || (uri != null && hiddenHashes.contains(localTypeKey(item) + ":" + hashByUri.get(uri)))) iterator.remove();
        }
    }

    private String localTypeKey(LocalMediaPhoto item) {
        return item != null && item.isVideo() ? "video" : "image";
    }

    private String cloudType(Photo item) {
        if (item == null) return "image";
        String type = item.getMediaType();
        if (type != null && type.toLowerCase(java.util.Locale.ROOT).startsWith("video")) return "video";
        String url = item.getImageUrl();
        if (url != null) {
            String path = Uri.parse(url).getPath();
            String extension = path == null ? "" : path.toLowerCase(java.util.Locale.ROOT);
            if (extension.endsWith(".mp4") || extension.endsWith(".mov")
                    || extension.endsWith(".m4v") || extension.endsWith(".webm")
                    || extension.endsWith(".3gp") || extension.endsWith(".mkv")) return "video";
        }
        return "image";
    }

    private LocalMediaPhoto copyLocalPhoto(LocalMediaPhoto source) {
        LocalMediaPhoto copy = new LocalMediaPhoto(source.getMediaId(), source.getContentUri(), source.getDisplayName(),
                source.getMimeType(), source.getSizeBytes(), source.getWidth(), source.getHeight(), source.getDateModified());
        copy.setDateTaken(source.getDateTaken());
        copy.setVideo(source.isVideo());
        copy.setDurationMs(source.getDurationMs());
        copy.setBackupStatus(source.getBackupStatus());
        return copy;
    }

    private String hashUri(android.content.Context context, String uriText) {
        if (uriText == null || uriText.isEmpty()) return null;
        try (InputStream input = context.getContentResolver().openInputStream(Uri.parse(uriText))) {
            if (input == null) return null;
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            byte[] buffer = new byte[32 * 1024];
            int count;
            while ((count = input.read(buffer)) != -1) digest.update(buffer, 0, count);
            StringBuilder hash = new StringBuilder(64);
            for (byte value : digest.digest()) hash.append(String.format(java.util.Locale.US, "%02x", value & 0xff));
            return hash.toString();
        } catch (Exception ignored) { return null; }
    }

    private void updateEmptyState() {
        boolean noMediaMatches = MediaTimelineUtils.filterAndSort(allDevicePhotos, currentMediaFilter,
                currentViewMode, currentSortMode, LocalMediaPhoto::getTimelineTimestamp,
                LocalMediaPhoto::getDisplayName, LocalMediaPhoto::isVideo).isEmpty();
        if (allDevicePhotos.isEmpty() || noMediaMatches) {
            binding.layoutEmpty.setVisibility(View.VISIBLE);
            binding.rvPhotos.setVisibility(View.GONE);
            binding.tvEmptyMessage.setText(cloudLoadFailed
                    ? "Không tải được ảnh cloud. Kiểm tra kết nối rồi thử làm mới."
                    : allDevicePhotos.isEmpty() ? getString(R.string.empty_photos) : getString(R.string.empty_media_filter));
        } else {
            binding.layoutEmpty.setVisibility(View.GONE);
            binding.rvPhotos.setVisibility(View.VISIBLE);
        }
    }

    @Override
    public void onLocalPhotoLongClick(LocalMediaPhoto photo, View view) {
        if (!selectionMode) selectionMode = true;
        if (!selectedPhotoKeys.contains(LocalPhotoAdapter.keyFor(photo))) togglePhotoSelection(photo);
        else renderSelectionMode();

    private void togglePhotoSelection(LocalMediaPhoto photo) {
        String key = LocalPhotoAdapter.keyFor(photo);
        if (!selectedPhotoKeys.add(key)) selectedPhotoKeys.remove(key);
        if (selectedPhotoKeys.isEmpty()) selectionMode = false;
        localPhotoAdapter.setSelectedKeys(selectedPhotoKeys);
        renderSelectionMode();
    }

    private void clearSelection() {
        selectionMode = false;
        selectedPhotoKeys.clear();
        if (localPhotoAdapter != null) localPhotoAdapter.setSelectedKeys(selectedPhotoKeys);
        renderSelectionMode();
    }

    private void renderSelectionMode() {
        if (binding == null) return;
        binding.layoutMultiSelect.setVisibility(selectionMode ? View.VISIBLE : View.GONE);
        binding.layoutHeaderActions.setVisibility(selectionMode ? View.GONE : View.VISIBLE);
        binding.tvHeaderTitle.setText(selectionMode ? "Đã chọn " + selectedPhotoKeys.size() : getString(R.string.tab_library));
        boolean anyCloud = false;
        for (LocalMediaPhoto photo : selectedPhotos()) if (photo.getCloudPhoto() != null) anyCloud = true;
        binding.btnSelectionAlbum.setEnabled(anyCloud);
        binding.btnSelectionFavorite.setEnabled(anyCloud);
        binding.btnSelectionAlbum.setAlpha(anyCloud ? 1f : 0.45f);
        binding.btnSelectionFavorite.setAlpha(anyCloud ? 1f : 0.45f);
        boolean allFavorites = anyCloud;
        for (LocalMediaPhoto photo : selectedPhotos()) {
            if (photo.getCloudPhoto() != null && !photo.getCloudPhoto().isFavorite()) allFavorites = false;
        }
        binding.btnSelectionFavorite.setText(allFavorites ? "Bỏ thích" : "Yêu thích");
    }

    private List<LocalMediaPhoto> selectedPhotos() {
        List<LocalMediaPhoto> result = new ArrayList<>();
        for (LocalMediaPhoto photo : allDevicePhotos) {
            if (selectedPhotoKeys.contains(LocalPhotoAdapter.keyFor(photo))) result.add(photo);
        }
        return result;
    }

    private void shareSelectedPhotos() {
        List<Uri> localUris = new ArrayList<>();
        List<String> cloudLinks = new ArrayList<>();
        for (LocalMediaPhoto photo : selectedPhotos()) {
            String local = photo.getContentUri();
            if (local != null && "content".equalsIgnoreCase(Uri.parse(local).getScheme())) localUris.add(Uri.parse(local));
            else if (photo.getCloudUrl() != null && !photo.getCloudUrl().isEmpty()) cloudLinks.add(photo.getCloudUrl());
        }
        if (localUris.isEmpty() && cloudLinks.isEmpty()) {
            Toast.makeText(requireContext(), "Không có ảnh hợp lệ để chia sẻ", Toast.LENGTH_SHORT).show();
            return;
        }
        Intent share;
        if (localUris.isEmpty()) {
            share = new Intent(Intent.ACTION_SEND);
            share.setType("text/plain");
            share.putExtra(Intent.EXTRA_TEXT, android.text.TextUtils.join("\n", cloudLinks));
        } else {
            share = new Intent(Intent.ACTION_SEND_MULTIPLE);
            share.setType("image/*");
            share.putParcelableArrayListExtra(Intent.EXTRA_STREAM, new ArrayList<>(localUris));
            ClipData clip = ClipData.newUri(requireContext().getContentResolver(), "PhotoVault images", localUris.get(0));
            for (int i = 1; i < localUris.size(); i++) clip.addItem(new ClipData.Item(localUris.get(i)));
            share.setClipData(clip);
            share.addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION);
            if (!cloudLinks.isEmpty()) share.putExtra(Intent.EXTRA_TEXT, android.text.TextUtils.join("\n", cloudLinks));
        }
        startActivity(Intent.createChooser(share, "Chia sẻ ảnh"));
    }

    private void applySelectedAlbum(String uid, String albumId) {
        List<Photo> cloudPhotos = new ArrayList<>();
        for (LocalMediaPhoto item : selectedPhotos()) if (item.getCloudPhoto() != null) cloudPhotos.add(item.getCloudPhoto());
        if (cloudPhotos.isEmpty()) return;
        AtomicInteger remaining = new AtomicInteger(cloudPhotos.size());
        AtomicInteger successes = new AtomicInteger();
        for (Photo photo : cloudPhotos) {
            List<String> ids = new ArrayList<>(photo.getAlbumIds());
            if (!ids.contains(albumId)) ids.add(albumId);
            else { if (remaining.decrementAndGet() == 0) finishAlbumSelection(successes.get(), cloudPhotos.size()); continue; }
            repository.updatePhotoAlbumsAsync(uid, photo.getId(), ids, new PhotoRepository.Callback<Void>() {
                @Override public void onSuccess(Void result) { photo.setAlbumIds(ids); successes.incrementAndGet(); finishOneAlbumUpdate(remaining, successes, cloudPhotos.size()); }
                @Override public void onError(String error) { finishOneAlbumUpdate(remaining, successes, cloudPhotos.size()); }
            });
        }
    }

    private void finishOneAlbumUpdate(AtomicInteger remaining, AtomicInteger successes, int total) {
        if (remaining.decrementAndGet() == 0) finishAlbumSelection(successes.get(), total);
    }

    private void finishAlbumSelection(int successes, int total) {
        if (binding == null || !isAdded()) return;
        Toast.makeText(requireContext(), "Đã thêm " + successes + "/" + total + " ảnh cloud vào album", Toast.LENGTH_LONG).show();
        clearSelection();
    }

    private void toggleSelectedFavorites() {
        UserInfo user = authRepository.getCurrentUser();
        if (user == null) return;
        List<Photo> cloudPhotos = new ArrayList<>();
        for (LocalMediaPhoto item : selectedPhotos()) if (item.getCloudPhoto() != null) cloudPhotos.add(item.getCloudPhoto());
        if (cloudPhotos.isEmpty()) return;
        AtomicInteger remaining = new AtomicInteger(cloudPhotos.size());
        AtomicInteger successes = new AtomicInteger();
        boolean favoriteValue = cloudPhotos.stream().allMatch(Photo::isFavorite) ? false : true;
        for (Photo photo : cloudPhotos) {
            boolean newFavorite = favoriteValue;
            repository.toggleFavoriteAsync(user.getId(), photo.getId(), newFavorite, new PhotoRepository.Callback<Void>() {
                @Override public void onSuccess(Void result) { photo.setFavorite(newFavorite); successes.incrementAndGet(); finishFavoriteUpdate(remaining, successes, cloudPhotos.size()); }
                @Override public void onError(String error) { finishFavoriteUpdate(remaining, successes, cloudPhotos.size()); }
            });
        }
    }

    private void finishFavoriteUpdate(AtomicInteger remaining, AtomicInteger successes, int total) {
        if (remaining.decrementAndGet() != 0 || binding == null || !isAdded()) return;
        Toast.makeText(requireContext(), "Đã cập nhật yêu thích cho " + successes.get() + "/" + total + " ảnh cloud", Toast.LENGTH_LONG).show();
        clearSelection();
    }

    private void moveSelectedToTrash(List<LocalMediaPhoto> selected) {
        if (trashOperationRunning || selected.isEmpty()) return;
        trashOperationRunning = true;
        binding.btnSelectionDelete.setEnabled(false);
        UserInfo user = authRepository.getCurrentUser();
        moveSelectedToTrashAt(selected, 0, user == null ? null : user.getId(), 0, new ArrayList<>());
    }

    private void moveSelectedToTrashAt(List<LocalMediaPhoto> selected, int index, String uid,
                                       int successes, List<String> failures) {
        if (index >= selected.size()) {
            trashOperationRunning = false;
            if (binding == null || !isAdded()) return;
            binding.btnSelectionDelete.setEnabled(true);
            if (selectedPhotoKeys.isEmpty()) clearSelection(); else renderSelectionMode();
            String summary = "Đã chuyển " + successes + "/" + selected.size() + " mục vào thùng rác.";
            if (failures.isEmpty()) Toast.makeText(requireContext(), summary, Toast.LENGTH_LONG).show();
            else new androidx.appcompat.app.AlertDialog.Builder(requireContext()).setTitle(summary)
                    .setMessage(android.text.TextUtils.join("\n\n", failures))
                    .setPositiveButton("Đóng", null).show();
            scanDeviceGallery();
            return;
        }
        LocalMediaPhoto item = selected.get(index);
        Photo target = photoForTrash(item);
        PhotoTrashManager.moveToTrash(requireContext(), uid, target, new PhotoTrashManager.Callback() {
            @Override public void onSuccess(Photo moved) {
                selectedPhotoKeys.remove(LocalPhotoAdapter.keyFor(item));
                if (binding != null && isAdded()) binding.getRoot().post(() -> moveSelectedToTrashAt(selected, index + 1, uid, successes + 1, failures));
            }
            @Override public void onError(String error) {
                failures.add((item.getDisplayName() == null ? "Mục" : item.getDisplayName()) + ": " + error);
                if (binding != null) binding.getRoot().post(() -> moveSelectedToTrashAt(selected, index + 1, uid, successes, failures));
            }
        });
    }

    private Photo photoForTrash(LocalMediaPhoto item) {
        Photo photo = item.getCloudPhoto() == null ? new Photo() : copyCloudPhoto(item.getCloudPhoto());
        if (item.getCloudPhoto() == null) {
            photo.setId("local:" + item.getMediaId());
            photo.setOwnerUid("local");
            photo.setImageUrl(item.getContentUri() == null ? "" : item.getContentUri());
            photo.setMediaType(item.isVideo() ? "video" : "image");
            photo.setCreatedAt(item.getTimelineTimestamp());
        }
        photo.setName(item.getDisplayName());
        if (item.getContentUri() != null) photo.setLocalMediaUri(item.getContentUri());
        if (item.getMediaId() > 0) photo.setLocalMediaId(item.getMediaId());
        photo.setLocalDateModified(item.getDateModified());
        return photo;
    }

    private void checkBackupOnboarding() {
        UserInfo user = authRepository.getCurrentUser();
        if (user != null && getContext() != null) {
            BackupPreferences prefs = new BackupPreferences(requireContext(), user.getId());
            if (!prefs.isOnboardingShown()) {
                BackupOnboardingDialog dialog = BackupOnboardingDialog.newInstance(new BackupOnboardingDialog.OnBackupDecisionListener() {
                    @Override
                    public void onEnableBackup(boolean isWifiOnly) {
                        Toast.makeText(getContext(), "Đã bật tự động sao lưu", Toast.LENGTH_SHORT).show();
                        UserInfo enabledUser = authRepository.getCurrentUser();
                        if (enabledUser != null && getContext() != null) {
                            BackupWorkScheduler.reconcile(requireContext(), enabledUser.getId(), true);
                        }
                    }

                    @Override
                    public void onSkipBackup() {
                        Toast.makeText(getContext(), "Đã bỏ qua sao lưu", Toast.LENGTH_SHORT).show();
                        UserInfo skippedUser = authRepository.getCurrentUser();
                        if (skippedUser != null && getContext() != null) {
                            BackupWorkScheduler.disable(requireContext(), skippedUser.getId());
                        }
                    }
                });
                dialog.show(getParentFragmentManager(), "backup_onboarding");
            }
        }
    }

    @Override

        super.onResume();
        if(binding!=null){
            android.content.SharedPreferences prefs=requireContext().getSharedPreferences("photovault_library",Context.MODE_PRIVATE);

        if (hasAnyMediaReadPermission()) scanDeviceGallery();
        else loadCloudPhotosForCurrentUser();
    }

        immediateBackupWork = null;
        periodicBackupWork = null;

    public static Intent createIntent(android.content.Context context, Photo current, List<Photo> collection) {
        Intent intent = new Intent(context, PhotoDetailActivity.class);
        intent.putExtra(EXTRA_PHOTO, current);
        ArrayList<Photo> safeCollection = new ArrayList<>();
        if (collection != null) safeCollection.addAll(collection);
        int index = 0;
        boolean found = false;
        for (int i = 0; i < safeCollection.size(); i++) {
            if (samePhoto(current, safeCollection.get(i))) { index = i; found = true; break; }
        }
        if (!found) { safeCollection.add(current); index = safeCollection.size() - 1; }
        String collectionToken = UUID.randomUUID().toString();
        COLLECTIONS.put(collectionToken, safeCollection);
        intent.putExtra(EXTRA_COLLECTION_TOKEN, collectionToken);
        intent.putExtra(EXTRA_PHOTO_INDEX, index);
        return intent;
    }

    private static boolean samePhoto(Photo first, Photo second) {
        if (first == null || second == null) return false;
        if (first.getId() != null && !first.getId().isEmpty() && first.getId().equals(second.getId())) return true;
        String firstLocal = first.getLocalMediaUri(), secondLocal = second.getLocalMediaUri();
        if (firstLocal != null && !firstLocal.isEmpty() && firstLocal.equals(secondLocal)) return true;
        String firstUrl = first.getImageUrl(), secondUrl = second.getImageUrl();
        return firstUrl != null && !firstUrl.isEmpty() && firstUrl.equals(secondUrl);
    }

    /** Converts an already merged local/cloud library item into the detail-screen model. */
    public static Photo photoForLocalMedia(LocalMediaPhoto media) {
        if (media == null) return new Photo();
        Photo result = media.getCloudPhoto() == null ? new Photo() : copyPhoto(media.getCloudPhoto());
        String localUri = media.getContentUri();
        if (media.getCloudPhoto() == null) {
            result.setId("local:" + (media.isVideo() ? "video:" : "image:") + media.getMediaId());
            result.setOwnerUid("local");
            result.setImageUrl(localUri == null ? "" : localUri);
            result.setDisplayUrl(localUri == null ? "" : localUri);
        }
        result.setName(media.getDisplayName() == null ? result.getName() : media.getDisplayName());
        result.setMediaType(media.isVideo() ? "video" : "image");
        if (localUri != null && !localUri.isEmpty()) result.setLocalMediaUri(localUri);
        if (media.getMediaId() > 0) result.setLocalMediaId(media.getMediaId());
        if (media.getDateModified() > 0) result.setLocalDateModified(media.getDateModified());
        if (media.getTimelineTimestamp() > 0) result.setCreatedAt(media.getTimelineTimestamp());
        return result;
    }

    @UnstableApi
    private void ensureVideoPlayer() {
        if (videoPlayer != null) return;
        videoPlayer = new ExoPlayer.Builder(this).build();
        binding.videoPlayerView.setPlayer(videoPlayer);
        videoPlayer.addListener(new Player.Listener() {
            @Override public void onPlaybackStateChanged(int state) {
                if (binding == null) return;
                binding.progressBar.setVisibility(state == Player.STATE_BUFFERING ? View.VISIBLE : View.GONE);
            }
            @Override public void onPlayerError(PlaybackException error) {
                if (binding == null) return;
                binding.progressBar.setVisibility(View.GONE);
                String cloudUrl = photo == null ? null : photo.getImageUrl();
                if (!videoCloudFallbackAttempted && cloudUrl != null && !cloudUrl.isEmpty()
                        && !cloudUrl.equals(activeVideoSource)) {
                    videoCloudFallbackAttempted = true;
                    activeVideoSource = cloudUrl;
                    binding.progressBar.setVisibility(View.VISIBLE);
                    videoPlayer.setMediaItem(MediaItem.fromUri(Uri.parse(cloudUrl)));
                    videoPlayer.prepare();
                    videoPlayer.setPlayWhenReady(true);
                    return;
                }
                Toast.makeText(PhotoDetailActivity.this,
                        "Không thể phát video: " + error.getErrorCodeName(), Toast.LENGTH_LONG).show();
            }
        });
    }

    private void toggleFavorite() {
        UserInfo user = authRepository.getCurrentUser();
        if (user == null) return;

        Photo targetPhoto = photo;
        boolean newFav = !targetPhoto.isFavorite();
        if (isLocalVideo(targetPhoto)) {
            Uri source = resolveMediaUri(targetPhoto);
            if (source == null) return;
            LocalVideoFavorites.set(this, user.getId(), source.toString(), newFav);
            targetPhoto.setFavorite(newFav);
            updateFavoriteIcon();
            Toast.makeText(this, newFav ? "Đã thêm video vào yêu thích" : "Đã bỏ video khỏi yêu thích", Toast.LENGTH_SHORT).show();
            return;
        }
        targetPhoto.setFavorite(newFav);
        updateFavoriteIcon();

        repository.toggleFavoriteAsync(user.getId(), targetPhoto.getId(), newFav, new PhotoRepository.Callback<Void>() {
            @Override
            public void onSuccess(Void result) {
                Toast.makeText(PhotoDetailActivity.this, newFav ? "Đã thêm vào yêu thích" : "Đã bỏ khỏi yêu thích", Toast.LENGTH_SHORT).show();
            }

            @Override
            public void onError(String error) {
                targetPhoto.setFavorite(!newFav);
                if (photo == targetPhoto) updateFavoriteIcon();
                Toast.makeText(PhotoDetailActivity.this, "Lỗi: " + error, Toast.LENGTH_SHORT).show();
            }
        });
    }

    private void sharePhoto() {
        Uri mediaUri = resolveMediaUri(photo);
        if (mediaUri == null) return;
        Intent shareIntent = new Intent(Intent.ACTION_SEND);
        shareIntent.putExtra(Intent.EXTRA_SUBJECT, photo.getName());
        if ("content".equalsIgnoreCase(mediaUri.getScheme())) {
            String fallbackType = isVideo(photo) ? "video/*" : "image/*";
            shareIntent.setType(getContentResolver().getType(mediaUri) != null ? getContentResolver().getType(mediaUri) : fallbackType);
            shareIntent.putExtra(Intent.EXTRA_STREAM, mediaUri);
            shareIntent.setClipData(android.content.ClipData.newUri(getContentResolver(), photo.getName(), mediaUri));
            shareIntent.addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION);
        } else {
            shareIntent.setType("text/plain");
            shareIntent.putExtra(Intent.EXTRA_TEXT, "Xem " + (isVideo(photo) ? "video" : "ảnh")
                    + " \"" + photo.getName() + "\" trên PhotoVault: " + mediaUri);
        }
        startActivity(Intent.createChooser(shareIntent, "Chia sẻ ảnh qua"));
    }

    private void loadUserAlbums() {
        if (photo.getOwnerUid() == null || "local".equals(photo.getOwnerUid())) return;
        UserInfo user = authRepository.getCurrentUser();
        if (user == null) return;

        repository.getAlbumsAsync(user.getId(), new PhotoRepository.Callback<List<Album>>() {
            @Override
            public void onSuccess(List<Album> albums) {
                userAlbums = albums != null ? albums : new ArrayList<>();
            }

            @Override
            public void onError(String error) {}
        });
    }

    private void showAddToAlbumDialog() {
        if (userAlbums == null || userAlbums.isEmpty()) {
            Toast.makeText(this, "Bạn chưa có album nào. Vui lòng tạo album trước!", Toast.LENGTH_LONG).show();
            return;
        }

        String[] albumNames = new String[userAlbums.size()];
        boolean[] checkedItems = new boolean[userAlbums.size()];

        for (int i = 0; i < userAlbums.size(); i++) {
            albumNames[i] = userAlbums.get(i).getName();
            checkedItems[i] = photo.getAlbumIds().contains(userAlbums.get(i).getId());
        }

        new AlertDialog.Builder(this)
                .setTitle("Thêm ảnh vào Album")
                .setMultiChoiceItems(albumNames, checkedItems, (dialog, which, isChecked) -> {
                    String albumId = userAlbums.get(which).getId();
                    if (isChecked) {
                        if (!photo.getAlbumIds().contains(albumId)) photo.getAlbumIds().add(albumId);
                    } else {
                        photo.getAlbumIds().remove(albumId);
                    }
                })
                .setPositiveButton("Lưu", (dialog, which) -> {
                    UserInfo user = authRepository.getCurrentUser();
                    if (user != null) {
                        repository.updatePhotoAlbumsAsync(user.getId(), photo.getId(), photo.getAlbumIds(), new PhotoRepository.Callback<Void>() {
                            @Override
                            public void onSuccess(Void result) {
                                Toast.makeText(PhotoDetailActivity.this, "Đã cập nhật album cho ảnh", Toast.LENGTH_SHORT).show();
                            }

                            @Override
                            public void onError(String error) {
                                Toast.makeText(PhotoDetailActivity.this, "Lỗi: " + error, Toast.LENGTH_SHORT).show();
                            }
                        });
                    }
                })
                .setNegativeButton("Hủy", null)
                .show();
    }

    private boolean canOverwriteImage() {
        if (imageUri == null || !"content".equalsIgnoreCase(imageUri.getScheme())) return false;

            String mime = getContentResolver().getType(imageUri);
            return "image/jpeg".equalsIgnoreCase(mime) || "image/jpg".equalsIgnoreCase(mime)
                    || "image/png".equalsIgnoreCase(mime) || "image/webp".equalsIgnoreCase(mime);

            return false;

    private void highlightPenColor(ImageButton selectedButton) {
        ImageButton[] buttons = {
                binding.colorRed, binding.colorYellow, binding.colorGreen, binding.colorBlue,
                binding.colorWhite, binding.colorBlack, binding.colorRosewood
        };
        int[] colors = {
                Color.parseColor("#EF4444"), Color.parseColor("#EAB308"), Color.parseColor("#22C55E"),
                Color.parseColor("#3B82F6"), Color.WHITE, Color.BLACK, Color.parseColor("#9F5967")
        };
        int accent = ContextCompat.getColor(this, R.color.editor_accent);
        int surface = ContextCompat.getColor(this, R.color.editor_surface);
        int border = ContextCompat.getColor(this, R.color.editor_border);
        int inset = Math.round(4f * getResources().getDisplayMetrics().density);
        int stroke = Math.max(1, Math.round(2f * getResources().getDisplayMetrics().density));

        for (int i = 0; i < buttons.length; i++) {
            ImageButton button = buttons[i];
            GradientDrawable swatch = new GradientDrawable();
            swatch.setShape(GradientDrawable.OVAL);
            swatch.setColor(colors[i]);
            button.setBackgroundTintList(null);
            boolean selected = button == selectedButton;
            if (selected) {
                GradientDrawable ring = new GradientDrawable();
                ring.setShape(GradientDrawable.OVAL);
                ring.setColor(surface);
                ring.setStroke(stroke, accent);
                button.setBackground(new LayerDrawable(new Drawable[]{ring, new InsetDrawable(swatch, inset)}));
                button.setElevation(2f * getResources().getDisplayMetrics().density);
            } else {
                swatch.setStroke(stroke, border);
                button.setBackground(swatch);
                button.setElevation(0f);
            }
            button.setSelected(selected);
        }
    }

    private void setupPreviewGestures() {
        binding.imgEditPreview.setScaleType(ImageView.ScaleType.MATRIX);
        binding.imgEditPreview.addOnLayoutChangeListener((view, left, top, right, bottom,
                                                          oldLeft, oldTop, oldRight, oldBottom) -> {
            if (right - left != oldRight - oldLeft || bottom - top != oldBottom - oldTop) {
                resetPreviewTransform();
            }
        });
        scaleGestureDetector = new ScaleGestureDetector(this, new ScaleGestureDetector.SimpleOnScaleGestureListener() {
            @Override public boolean onScale(ScaleGestureDetector detector) {
                float targetScale = Math.max(minimumPreviewScale,
                        Math.min(minimumPreviewScale * 5f, previewScale * detector.getScaleFactor()));
                float factor = targetScale / previewScale;
                previewMatrix.postScale(factor, factor, detector.getFocusX(), detector.getFocusY());
                previewScale = targetScale;
                applyPreviewMatrix();
                return true;
            }
        });
        binding.imgEditPreview.setOnTouchListener((view, event) -> {
            if (overlaysAreActive()) return false;
            scaleGestureDetector.onTouchEvent(event);
            switch (event.getActionMasked()) {
                case MotionEvent.ACTION_DOWN:
                    lastPanX = event.getX();
                    lastPanY = event.getY();
                    return true;
                case MotionEvent.ACTION_MOVE:
                    if (event.getPointerCount() == 1 && previewScale > minimumPreviewScale * 1.001f
                            && !scaleGestureDetector.isInProgress()) {
                        previewMatrix.postTranslate(event.getX() - lastPanX, event.getY() - lastPanY);
                        applyPreviewMatrix();
                    }
                    lastPanX = event.getX();
                    lastPanY = event.getY();
                    return true;
                case MotionEvent.ACTION_UP:
                case MotionEvent.ACTION_CANCEL:
                    return true;
                default:
                    return true;
            }
        });
        // Let a two-finger gesture zoom the image beneath crop/draw overlays. Their own
        // touch handlers cancel the active one-finger gesture once a second pointer arrives.
        binding.cropOverlay.setOnTouchListener((view, event) -> {
            handleOverlayTransformGesture(event);
            return false;
        });
        binding.drawingOverlay.setOnTouchListener((view, event) -> {
            handleOverlayTransformGesture(event);
            return false;
        });
        binding.btnFit.setOnClickListener(v -> resetPreviewTransform());
    }

    private boolean overlaysAreActive() {
        return binding.cropOverlay.getVisibility() == View.VISIBLE
                || binding.drawingOverlay.getVisibility() == View.VISIBLE;
    }

    private void handleOverlayTransformGesture(MotionEvent event) {
        scaleGestureDetector.onTouchEvent(event);
        int action = event.getActionMasked();
        if (action == MotionEvent.ACTION_POINTER_DOWN) {
            hasOverlayFocus = false;
        } else if (event.getPointerCount() >= 2 && action == MotionEvent.ACTION_MOVE) {
            float focusX = 0f, focusY = 0f;
            for (int i = 0; i < event.getPointerCount(); i++) {
                focusX += event.getX(i);
                focusY += event.getY(i);
            }
            focusX /= event.getPointerCount();
            focusY /= event.getPointerCount();
            if (hasOverlayFocus) {
                previewMatrix.postTranslate(focusX - lastOverlayFocusX, focusY - lastOverlayFocusY);
                applyPreviewMatrix();
            }
            lastOverlayFocusX = focusX;
            lastOverlayFocusY = focusY;
            hasOverlayFocus = true;
        } else if (action == MotionEvent.ACTION_POINTER_UP || action == MotionEvent.ACTION_UP
                || action == MotionEvent.ACTION_CANCEL) {
            hasOverlayFocus = false;
        }
    }

    private void resetPreviewTransform() {
        if (binding == null || binding.imgEditPreview.getDrawable() == null) return;
        binding.imgEditPreview.post(() -> {
            if (binding == null || binding.imgEditPreview.getDrawable() == null) return;
            int imageWidth = binding.imgEditPreview.getDrawable().getIntrinsicWidth();
            int imageHeight = binding.imgEditPreview.getDrawable().getIntrinsicHeight();
            int viewWidth = binding.imgEditPreview.getWidth();
            int viewHeight = binding.imgEditPreview.getHeight();
            if (imageWidth <= 0 || imageHeight <= 0 || viewWidth <= 0 || viewHeight <= 0) return;
            minimumPreviewScale = Math.min((float) viewWidth / imageWidth, (float) viewHeight / imageHeight);
            previewScale = minimumPreviewScale;
            float left = (viewWidth - imageWidth * minimumPreviewScale) / 2f;
            float top = (viewHeight - imageHeight * minimumPreviewScale) / 2f;
            previewMatrix.setScale(minimumPreviewScale, minimumPreviewScale);
            previewMatrix.postTranslate(left, top);
            applyPreviewMatrix();
        });
    }

    private void applyPreviewMatrix() {
        if (binding == null) return;
        clampPreviewTranslation();
        binding.imgEditPreview.setImageMatrix(previewMatrix);
        updateImageBoundsForOverlays();
    }

    private void clampPreviewTranslation() {
        if (binding == null || binding.imgEditPreview.getDrawable() == null) return;
        float[] values = new float[9];
        previewMatrix.getValues(values);
        float width = binding.imgEditPreview.getDrawable().getIntrinsicWidth() * values[Matrix.MSCALE_X];
        float height = binding.imgEditPreview.getDrawable().getIntrinsicHeight() * values[Matrix.MSCALE_Y];
        float viewWidth = binding.imgEditPreview.getWidth();
        float viewHeight = binding.imgEditPreview.getHeight();
        float minX = width > viewWidth ? viewWidth - width : (viewWidth - width) / 2f;
        float maxX = width > viewWidth ? 0f : minX;
        float minY = height > viewHeight ? viewHeight - height : (viewHeight - height) / 2f;
        float maxY = height > viewHeight ? 0f : minY;
        float dx = Math.max(minX, Math.min(maxX, values[Matrix.MTRANS_X])) - values[Matrix.MTRANS_X];
        float dy = Math.max(minY, Math.min(maxY, values[Matrix.MTRANS_Y])) - values[Matrix.MTRANS_Y];
        previewMatrix.postTranslate(dx, dy);
    }

    private RectF getImageDisplayBounds(ImageView imageView) {
        if (imageView == null || imageView.getDrawable() == null) return null;

        Drawable d = imageView.getDrawable();
        int origW = d.getIntrinsicWidth();
        int origH = d.getIntrinsicHeight();
        if (origW <= 0 || origH <= 0) return null;
        RectF bounds = new RectF(0f, 0f, origW, origH);
        imageView.getImageMatrix().mapRect(bounds);
        return bounds;
    }

    private void executeCrop() {
        if (currentBitmap == null || pendingPreviewRender != null || renderingPreview) return;
        Rect cropRect = binding.cropOverlay.getCroppedBitmapArea(currentBitmap, binding.imgEditPreview);
        if (cropRect != null && cropRect.width() > 10 && cropRect.height() > 10) {
            pushUndoState();
            RectF normalized = new RectF((float) cropRect.left / currentBitmap.getWidth(),
                    (float) cropRect.top / currentBitmap.getHeight(),
                    (float) cropRect.right / currentBitmap.getWidth(),
                    (float) cropRect.bottom / currentBitmap.getHeight());
            editState.operations.add(PhotoEditState.Operation.crop(normalized));
            markEdited();
            binding.cropOverlay.setVisibility(View.GONE);
            schedulePreviewRender(true);
            Toast.makeText(this, "Đã cắt ảnh!", Toast.LENGTH_SHORT).show();
        } else {
            Toast.makeText(this, "Vùng cắt không hợp lệ", Toast.LENGTH_SHORT).show();
        }
    }

    private void applyCropRatio(float ratioW, float ratioH) {
        if (currentBitmap == null) return;
        RectF bounds = getImageDisplayBounds(binding.imgEditPreview);
        if (bounds == null) return;

        float targetRatio = ratioW / ratioH;
        float boundsRatio = bounds.width() / bounds.height();

        float cropW = bounds.width();
        float cropH = bounds.height();

        if (boundsRatio > targetRatio) {
            cropW = bounds.height() * targetRatio;
        } else {
            cropH = bounds.width() / targetRatio;
        }

        float left = bounds.left + (bounds.width() - cropW) / 2f;
        float top = bounds.top + (bounds.height() - cropH) / 2f;

        RectF newRect = new RectF(left, top, left + cropW, top + cropH);
        binding.cropOverlay.getCropRect().set(newRect);
        binding.cropOverlay.invalidate();
    }

    private void loadImageBitmap() {
        binding.progressBar.setVisibility(View.VISIBLE);

        Glide.with(this)
                .asBitmap()
                .load(imageUri)
                .override(1800, 1800)
                .centerInside()
                .into(new CustomTarget<Bitmap>() {
                    @Override
                    public void onResourceReady(@NonNull Bitmap resource, @Nullable Transition<? super Bitmap> transition) {
                        binding.progressBar.setVisibility(View.GONE);
                        originalBitmap = resource.copy(Bitmap.Config.ARGB_8888, true);
                        adjustmentSettings = editState.adjustments;
                        updateFilterThumbnails();
                        schedulePreviewRender(true);
                    }

                    @Override
                    public void onLoadCleared(@Nullable Drawable placeholder) {}

                    @Override
                    public void onLoadFailed(@Nullable Drawable errorDrawable) {
                        binding.progressBar.setVisibility(View.GONE);
                        Toast.makeText(PhotoEditorActivity.this, "Lỗi tải ảnh để chỉnh sửa", Toast.LENGTH_SHORT).show();
                        finish();
                    }
                });
    }

        else if (filterType == BitmapEditor.FILTER_DRAMATIC) binding.chipGroupFilters.check(binding.filterSepia.getId());
        else if (filterType == BitmapEditor.FILTER_FADE) binding.chipGroupFilters.check(binding.filterFade.getId());

    private void restoreEditedPreviewAfterCompare() {
        if (!comparingOriginal || binding == null) return;

        binding.drawingOverlay.setVisibility(compareDrawVisibility);

                || currentBitmap.getHeight() != compareBitmapHeight)) resetPreviewTransform();
        else binding.imgEditPreview.setImageMatrix(compareSavedMatrix);

    private void showBitmapFitForCompare(Bitmap bitmap) {
        if (binding == null || bitmap == null) return;
        int viewWidth = binding.imgEditPreview.getWidth();
        int viewHeight = binding.imgEditPreview.getHeight();
        if (viewWidth <= 0 || viewHeight <= 0) {
            binding.imgEditPreview.post(() -> showBitmapFitForCompare(bitmap));
            return;
        }
        float scale = Math.min((float) viewWidth / bitmap.getWidth(), (float) viewHeight / bitmap.getHeight());
        Matrix matrix = new Matrix();
        matrix.setScale(scale, scale);
        matrix.postTranslate((viewWidth - bitmap.getWidth() * scale) / 2f,
                (viewHeight - bitmap.getHeight() * scale) / 2f);
        binding.imgEditPreview.setImageMatrix(matrix);
    }

    private void pushUndoState() {
        if (restoringState) return;
        undoStates.push(editState.copy());
        while (undoStates.size() > 40) undoStates.removeLast();
        redoStates.clear();
    }

    private void markEdited() {
        hasUnsavedEdits = !editState.isNeutral();
    }

    private void undoEdit() {
        if (undoStates.isEmpty()) return;
        redoStates.push(editState.copy());
        editState = undoStates.pop();
        restoreEditState();
    }

    private void redoEdit() {
        if (redoStates.isEmpty()) return;
        undoStates.push(editState.copy());
        editState = redoStates.pop();
        restoreEditState();
    }

    private void restoreEditState() {
        restoringState = true;
        adjustmentSettings = editState.adjustments;
        selectedFilter = editState.filter;
        binding.sbExposure.setProgress(Math.round(300 + adjustmentSettings.getExposure() * 100));
        binding.sbBrightness.setProgress(Math.round(100 + adjustmentSettings.getBrightness()));
        binding.sbHighlights.setProgress(Math.round(100 + adjustmentSettings.getHighlights()));
        binding.sbShadows.setProgress(Math.round(100 + adjustmentSettings.getShadows()));
        binding.sbContrast.setProgress(Math.round(adjustmentSettings.getContrast() * 100));
        binding.sbSaturation.setProgress(Math.round(adjustmentSettings.getSaturation() * 100));
        binding.sbTemperature.setProgress(Math.round(100 + adjustmentSettings.getTemperature()));
        binding.sbTint.setProgress(Math.round(100 + adjustmentSettings.getTint()));
        binding.sbFilterIntensity.setProgress(Math.round(editState.filterIntensity * 100));
        syncFilterSelection();
        binding.drawingOverlay.clear();
        markEdited();
        restoringState = false;
        schedulePreviewRender(true);
    }

    private void syncFilterSelection() {
        int checked = -1;
        switch (editState.filter) {
            case BitmapEditor.FILTER_VIVID: checked = binding.filterVivid.getId(); break;
            case BitmapEditor.FILTER_WARM: checked = binding.filterWarm.getId(); break;
            case BitmapEditor.FILTER_COOL: checked = binding.filterCool.getId(); break;
            case BitmapEditor.FILTER_MONO: checked = binding.filterGrayscale.getId(); break;
            case BitmapEditor.FILTER_DRAMATIC: checked = binding.filterSepia.getId(); break;
            case BitmapEditor.FILTER_FADE: checked = binding.filterFade.getId(); break;
            default: checked = binding.filterOriginal.getId(); break;
        }
        binding.chipGroupFilters.check(checked);
    }

    private Bitmap renderEditRecipe(Bitmap source, PhotoEditState state) {
        if (source == null) return null;
        Bitmap working = source;
        boolean ownsWorking = false;
        Bitmap strokeLayer = null;
        for (PhotoEditState.Operation operation : state.operations) {
            if (Thread.currentThread().isInterrupted()) {
                if (strokeLayer != null && !strokeLayer.isRecycled()) strokeLayer.recycle();
                if (ownsWorking && !working.isRecycled()) working.recycle();
                return null;
            }
            Bitmap next = working;
            if (operation.type != PhotoEditState.STROKE && strokeLayer != null) {
                next = working.copy(Bitmap.Config.ARGB_8888, true);
                new android.graphics.Canvas(next).drawBitmap(strokeLayer, 0f, 0f, null);
                strokeLayer.recycle(); strokeLayer = null;
                if (ownsWorking && !working.isRecycled()) working.recycle();
                working = next; ownsWorking = true;
            }
            if (operation.type == PhotoEditState.ROTATE) next = BitmapEditor.rotate(working, operation.value);
            else if (operation.type == PhotoEditState.FLIP) next = BitmapEditor.flip(working, operation.horizontal, operation.vertical);
            else if (operation.type == PhotoEditState.CROP) next = BitmapEditor.cropNormalized(working, operation.crop);
            else if (operation.type == PhotoEditState.STROKE) {
                if (strokeLayer == null) strokeLayer = Bitmap.createBitmap(working.getWidth(), working.getHeight(), Bitmap.Config.ARGB_8888);
                BitmapEditor.drawNormalizedStroke(strokeLayer, operation.points, operation.color, operation.strokeWidth, operation.erase);
                continue;
            }
            if (next != working) {
                if (ownsWorking && !working.isRecycled()) working.recycle();
                working = next;
                ownsWorking = true;
            }
        }
        if (strokeLayer != null) {
            Bitmap next = working.copy(Bitmap.Config.ARGB_8888, true);
            new android.graphics.Canvas(next).drawBitmap(strokeLayer, 0f, 0f, null);
            strokeLayer.recycle();
            if (ownsWorking && !working.isRecycled()) working.recycle();
            working = next; ownsWorking = true;
        }
        if (Thread.currentThread().isInterrupted()) {
            if (ownsWorking && !working.isRecycled()) working.recycle();
            return null;
        }
        Bitmap output = BitmapEditor.renderColor(working, state.adjustments, state.filter, state.filterIntensity);
        if (ownsWorking && output != working && !working.isRecycled()) working.recycle();
        return output;
    }

    private void schedulePreviewRender(boolean resetTransform) {
        if (originalBitmap == null || binding == null) return;
        if (pendingPreviewRender != null) mainHandler.removeCallbacks(pendingPreviewRender);
        if (previewFuture != null) previewFuture.cancel(true);
        int version = renderVersion.incrementAndGet();
        PhotoEditState snapshot = editState.copy();
        Bitmap source = originalBitmap;
        pendingPreviewRender = () -> {
            pendingPreviewRender = null;
            renderingPreview = true;
            previewFuture = renderExecutor.submit(() -> {
                Bitmap result = renderEditRecipe(source, snapshot);
                mainHandler.post(() -> {
                    if (version != renderVersion.get() || binding == null) {
                        if (result != null && !result.isRecycled()) result.recycle();
                        return;
                    }
                    renderingPreview = false;
                    if (result == null) return;
                    Bitmap previous = currentBitmap;
                    currentBitmap = result;
                    if (!comparingOriginal) {
                        binding.imgEditPreview.setImageBitmap(result);
                        if (resetTransform) resetPreviewTransform();
                        else updateImageBoundsForOverlays();
                    }
                    if (previous != null && previous != originalBitmap && previous != result && !previous.isRecycled()) previous.recycle();
                });
            });
        };
        mainHandler.postDelayed(pendingPreviewRender, 55);
    }

    private void updateFilterThumbnails() {
        if (originalBitmap == null) return;
        Bitmap sample = Bitmap.createScaledBitmap(originalBitmap, 52, 52, true);
        setFilterThumbnail(binding.filterOriginal, BitmapEditor.FILTER_ORIGINAL, sample);
        setFilterThumbnail(binding.filterVivid, BitmapEditor.FILTER_VIVID, sample);
        setFilterThumbnail(binding.filterWarm, BitmapEditor.FILTER_WARM, sample);
        setFilterThumbnail(binding.filterCool, BitmapEditor.FILTER_COOL, sample);
        setFilterThumbnail(binding.filterGrayscale, BitmapEditor.FILTER_MONO, sample);
        setFilterThumbnail(binding.filterSepia, BitmapEditor.FILTER_DRAMATIC, sample);
        setFilterThumbnail(binding.filterFade, BitmapEditor.FILTER_FADE, sample);
        if (sample != originalBitmap && !sample.isRecycled()) sample.recycle();
    }

    private void setFilterThumbnail(com.google.android.material.chip.Chip chip, int filter, Bitmap sample) {
        Bitmap thumbnail = filter == BitmapEditor.FILTER_ORIGINAL
                ? sample.copy(Bitmap.Config.ARGB_8888, false)
                : BitmapEditor.renderColor(sample, new AdjustmentSettings(), filter, 1f);
        chip.setChipIcon(new BitmapDrawable(getResources(), thumbnail));
        chip.setChipIconVisible(true);
        chip.setChipIconSize(getResources().getDisplayMetrics().density * 28f);
    }

    private void saveCopy() { saveEditedImage(false); }

    private void saveEditedImage(boolean overwrite) {
        if (saveInProgress || currentBitmap == null) return;
        saveInProgress = true;
        PhotoEditState snapshot = editState.copy();
        setSavingUi(true);
        renderExecutor.execute(() -> {
            FutureTarget<Bitmap> sourceTarget = null;
            Bitmap output;
            try {
                sourceTarget = Glide.with(getApplicationContext()).asBitmap().load(imageUri)
                        .submit(Target.SIZE_ORIGINAL, Target.SIZE_ORIGINAL);
                Bitmap fullSource = sourceTarget.get();
                output = renderEditRecipe(fullSource, snapshot);
                if (output == fullSource) output = fullSource.copy(Bitmap.Config.ARGB_8888, true);
                Glide.with(getApplicationContext()).clear(sourceTarget);
                if (output == null) throw new IllegalStateException("Không thể kết xuất ảnh đã chỉnh sửa.");
            } catch (Exception error) {
                if (sourceTarget != null) Glide.with(getApplicationContext()).clear(sourceTarget);
                mainHandler.post(() -> finishSaveWithError(error.getMessage()));
                return;
            } catch (OutOfMemoryError exhausted) {
                if (sourceTarget != null) Glide.with(getApplicationContext()).clear(sourceTarget);
                mainHandler.post(() -> finishSaveWithError("Ảnh quá lớn để xuất an toàn trên bộ nhớ hiện có. Hãy thử ảnh nhỏ hơn hoặc giảm thao tác."));
                return;
            }
            final Bitmap renderedOutput = output;
            ImageSaveUtils.SaveCallback callback = new ImageSaveUtils.SaveCallback() {
                private void releaseOutput() { if (!renderedOutput.isRecycled()) renderedOutput.recycle(); }
                @Override public void onSuccess(Uri savedUri, String filePath) {
                    releaseOutput();
                    if (binding == null) return;
                    Intent result = new Intent();
                    result.setData(savedUri);
                    result.putExtra("saved_uri", savedUri.toString());
                    result.putExtra("saved_name", filePath);
                    setResult(RESULT_OK, result);
                    Toast.makeText(PhotoEditorActivity.this,
                            overwrite ? "Đã cập nhật ảnh gốc." : "Đã lưu bản sao vào thư viện.", Toast.LENGTH_LONG).show();
                    finish();
                }
                @Override public void onError(String error) {
                    releaseOutput();
                    finishSaveWithError(error);
                }
                @Override public void onPermissionRequired(IntentSender sender) {
                    releaseOutput();
                    saveInProgress = false;
                    setSavingUi(false);
                    try {
                        overwritePermissionLauncher.launch(new IntentSenderRequest.Builder(sender).build());
                    } catch (Exception launchError) {
                        Toast.makeText(PhotoEditorActivity.this,
                                "Không mở được yêu cầu quyền sửa ảnh. Ảnh gốc chưa được đổi; hãy lưu bản sao.", Toast.LENGTH_LONG).show();
                    }
                }
            };
            if (overwrite) ImageSaveUtils.overwriteOriginalBitmap(this, renderedOutput, imageUri, callback);
            else ImageSaveUtils.saveBitmapAsCopy(this, renderedOutput, callback);
        });
    }

    private void finishSaveWithError(String message) {
        saveInProgress = false;
        if (binding == null) return;
        setSavingUi(false);
        Toast.makeText(this, "Không lưu được ảnh: " + message, Toast.LENGTH_LONG).show();
    }

    private void setSavingUi(boolean saving) {
        if (binding == null) return;
        binding.progressBar.setVisibility(saving ? View.VISIBLE : View.GONE);
        setChildrenEnabled(binding.topBar, !saving);
        setChildrenEnabled(binding.bottomControlsContainer, !saving);
        binding.previewContainer.setEnabled(!saving);
        binding.imgEditPreview.setEnabled(!saving);
        binding.cropOverlay.setEnabled(!saving);
        binding.drawingOverlay.setEnabled(!saving);
        if (!saving) {
            binding.btnOverwrite.setEnabled(canOverwriteImage());
            binding.btnOverwrite.setAlpha(binding.btnOverwrite.isEnabled() ? 1f : 0.5f);
        }
    }

    private void setChildrenEnabled(View view, boolean enabled) {
        view.setEnabled(enabled);
        if (view instanceof ViewGroup) {
            ViewGroup group = (ViewGroup) view;
            for (int i = 0; i < group.getChildCount(); i++) setChildrenEnabled(group.getChildAt(i), enabled);
        }
    }

    private void confirmOverwrite() {
        if (!canOverwriteImage()) {
            Toast.makeText(this, "Chỉ có thể ghi đè ảnh gốc trong thư viện thiết bị.", Toast.LENGTH_LONG).show();
            return;
        }
        new AlertDialog.Builder(this)
                .setTitle("Ghi đè ảnh gốc?")
                .setMessage("Thao tác này sẽ thay thế nội dung ảnh hiện tại bằng phiên bản đã chỉnh sửa. Bạn có muốn tiếp tục?")
                .setPositiveButton("Tiếp tục", (dialog, which) -> requestOverwritePermission())
                .setNegativeButton("Hủy", null)
                .show();
    }

    private void requestOverwritePermission() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {
            try {
                PendingIntent request = android.provider.MediaStore.createWriteRequest(
                        getContentResolver(), Collections.singletonList(imageUri));
                overwritePermissionLauncher.launch(new IntentSenderRequest.Builder(request.getIntentSender()).build());
            } catch (Exception error) {
                Toast.makeText(this, "Không thể xin quyền ghi ảnh này. Hãy lưu bản sao.", Toast.LENGTH_LONG).show();
            }
        } else {
            overwriteOriginal();
        }
    }

    private void overwriteOriginal() { saveEditedImage(true); }

    private void confirmCancel() {
        if (!hasUnsavedEdits && !binding.drawingOverlay.hasPaths()) {
            finish();
            return;
        }
        new AlertDialog.Builder(this)
                .setTitle("Hủy chỉnh sửa?")
                .setMessage("Các thay đổi chưa lưu sẽ bị mất. Bạn có muốn thoát?")
                .setPositiveButton("Bỏ thay đổi", (dialog, which) -> finish())
                .setNegativeButton("Hủy", null)
                .setNeutralButton("Tiếp tục chỉnh sửa", null)
                .show();
    }

    private void showSaveOptions() {
        String[] options = binding.btnOverwrite.isEnabled()
                ? new String[]{"Lưu bản sao", "Ghi đè ảnh gốc"}
                : new String[]{"Lưu bản sao"};
        new AlertDialog.Builder(this)
                .setTitle("Lưu chỉnh sửa")
                .setItems(options, (dialog, which) -> {
                    if (which == 0) saveCopy(); else confirmOverwrite();
                })
                .setNegativeButton("Tiếp tục chỉnh sửa", null)
                .show();
    }

    private boolean hasUnsavedChanges() {
        return hasUnsavedEdits || (binding != null && binding.drawingOverlay.hasPaths());
    }

    @Override
    protected void onDestroy() {
        renderVersion.incrementAndGet();
        if (pendingPreviewRender != null) mainHandler.removeCallbacks(pendingPreviewRender);
        if (previewFuture != null) previewFuture.cancel(true);
        renderExecutor.shutdownNow();
        if (binding != null) binding = null;
        super.onDestroy();
    }

    @Override
    @SuppressWarnings("deprecation")
    public void onBackPressed() {
        if (hasUnsavedChanges()) confirmCancel();
        else super.onBackPressed();
    }
}

        requestPermissionLauncher = registerForActivityResult(new ActivityResultContracts.RequestMultiplePermissions(), grants -> {

            if (granted) {

                if (currentUser != null && (scanAfterPermission || prefs.isBackupEnabled())) {

        });
        setupUI();
        setupListeners();
        observeWork();
        if (prefs.isBackupEnabled()) BackupWorkScheduler.reconcile(this, user.getId(), true);
        checkPermissionAndLoad();

    private void checkPermissionAndLoad() {
        if (hasMediaPermission()) {
            loadBackupStats();
            return;
        }
        if (Build.VERSION.SDK_INT >= 34) {
            requestPermissionLauncher.launch(new String[]{Manifest.permission.READ_MEDIA_IMAGES,
                    Manifest.permission.READ_MEDIA_VIDEO, Manifest.permission.READ_MEDIA_VISUAL_USER_SELECTED});
        } else if (Build.VERSION.SDK_INT >= 33) {
            requestPermissionLauncher.launch(new String[]{Manifest.permission.READ_MEDIA_IMAGES, Manifest.permission.READ_MEDIA_VIDEO});
        } else {
            requestPermissionLauncher.launch(new String[]{Manifest.permission.READ_EXTERNAL_STORAGE});
        }
    }

    private boolean hasMediaPermission() {
        if (Build.VERSION.SDK_INT >= 34) {
            return (ContextCompat.checkSelfPermission(this, Manifest.permission.READ_MEDIA_IMAGES) == PackageManager.PERMISSION_GRANTED
                    && ContextCompat.checkSelfPermission(this, Manifest.permission.READ_MEDIA_VIDEO) == PackageManager.PERMISSION_GRANTED)
                    || ContextCompat.checkSelfPermission(this, Manifest.permission.READ_MEDIA_VISUAL_USER_SELECTED) == PackageManager.PERMISSION_GRANTED;
        }
        if (Build.VERSION.SDK_INT >= 33) {
            return ContextCompat.checkSelfPermission(this, Manifest.permission.READ_MEDIA_IMAGES) == PackageManager.PERMISSION_GRANTED
                    && ContextCompat.checkSelfPermission(this, Manifest.permission.READ_MEDIA_VIDEO) == PackageManager.PERMISSION_GRANTED;
        }
        return ContextCompat.checkSelfPermission(this, Manifest.permission.READ_EXTERNAL_STORAGE) == PackageManager.PERMISSION_GRANTED;
    }

    private void startBackup() {
        if (!binding.switchAutoBackup.isChecked()) binding.switchAutoBackup.setChecked(true);
        else checkPermissionAndScanQueue();

    private void checkPermissionAndScanQueue() {
        if (!hasMediaPermission()) {
            scanAfterPermission = true;
            checkPermissionAndLoad();
            return;
        }
        scanQueueAndEnqueue(true);
    }

    private void loadBackupStats() { scanQueueAndEnqueue(false); }

    private void scanQueueAndEnqueue(boolean startBackup) {
        UserInfo user = authRepository.getCurrentUser();
        if (user == null) return;
        binding.progressBar.setVisibility(View.VISIBLE);
        backupManager.processBackupState(this, user.getId(), new PhotoBackupManager.ManagerCallback() {
            @Override public void onStatsReady(PhotoBackupManager.BackupStats stats, List<LocalMediaPhoto> all) {
                latestStats = stats;
                binding.progressBar.setVisibility(View.GONE);
                binding.tvStatTotalLocal.setText(String.valueOf(stats.totalLocalPhotos));
                binding.tvStatBackedUp.setText(String.valueOf(stats.backedUpCount));
                binding.tvStatRemaining.setText(String.valueOf(stats.remainingCount));
                binding.tvStatUploading.setText("0");
                renderMediaTypeStats(0, 0, null);
                binding.tvBackupDetail.setText("Số media được đọc từ thiết bị hiện tại.");
                binding.pbBackupProgress.setProgress(stats.totalLocalPhotos == 0 ? 0
                        : stats.backedUpCount * 100 / stats.totalLocalPhotos);
                if (startBackup) {
                    if (stats.pendingPhotos.isEmpty()) {
                        Toast.makeText(BackupCenterActivity.this,
                                stats.totalLocalPhotos == 0 ? "Không tìm thấy media trên thiết bị." : "Tất cả media đã được sao lưu.",
                                Toast.LENGTH_LONG).show();
                    } else {
                        Toast.makeText(BackupCenterActivity.this, "Đang đưa " + stats.pendingPhotos.size() + " media vào hàng đợi backup.", Toast.LENGTH_SHORT).show();
                        enqueueBackup();
                    }
                }
            }
            @Override public void onError(String error) {
                binding.progressBar.setVisibility(View.GONE);
                Toast.makeText(BackupCenterActivity.this, "Lỗi quét: " + error, Toast.LENGTH_LONG).show();
            }
        });
    }

    private void enqueueBackup() {
        if (!prefs.isBackupEnabled() || prefs.isPaused()) return;
        UserInfo user = authRepository.getCurrentUser();
        if (user != null) BackupWorkScheduler.enqueueIfEligible(this, user.getId());
    }

    private void observeWork() {
        UserInfo user = authRepository.getCurrentUser();
        if (user == null) return;
        WorkManager workManager = WorkManager.getInstance(this);
        workManager.getWorkInfosForUniqueWorkLiveData(BackupWorkScheduler.immediateWorkName(user.getId()))
                .observe(this, infos -> updateWorkInfo(infos, false));
        workManager.getWorkInfosForUniqueWorkLiveData(BackupWorkScheduler.periodicWorkName(user.getId()))
                .observe(this, infos -> updateWorkInfo(infos, true));
    }

    private void updateWorkInfo(List<WorkInfo> infos, boolean periodic) {
        if (infos == null || infos.isEmpty() || binding == null) return;
        WorkInfo info = infos.get(0);
        if (info.getState() == WorkInfo.State.RUNNING) {
            if (periodic) periodicWorkWasRunning = true;
            binding.progressBar.setVisibility(View.VISIBLE);
            Data progress = info.getProgress();
            int uploading = progress.getInt("uploading", 0);
            int failures = progress.getInt("failures", 0);
            int uploadedImages = progress.getInt("uploaded_images", 0);
            int uploadedVideos = progress.getInt("uploaded_videos", 0);
            String uploadingType = uploading > 0 ? progress.getString("current_type") : null;
            renderMediaTypeStats(uploadedImages, uploadedVideos, uploadingType);
            binding.tvStatUploading.setText(String.valueOf(uploading));
            String current = progress.getString("current_name");
            binding.tvBackupDetail.setText((current == null ? "Đang xử lý media" : "Đang tải: " + current)
                    + " • Lỗi trong lượt này: " + failures);
        } else if (info.getState().isFinished()) {
            loadCloudStorageUsage();
            binding.progressBar.setVisibility(View.GONE);
            binding.tvStatUploading.setText("0");
            if (info.getState() == WorkInfo.State.SUCCEEDED) {
                Data data = info.getOutputData();
                renderMediaTypeStats(data.getInt("uploaded_images", 0), data.getInt("uploaded_videos", 0), null);
                int success = data.getInt("success", 0), failures = data.getInt("failures", 0);
                if (success > 0 || failures > 0) {
                    String error = data.getString("error");
                    String message = "Hoàn tất: " + success + " media thành công, " + failures + " media lỗi.";
                    if (failures > 0 && error != null && !error.isEmpty()) message += " " + error;
                    Toast.makeText(this, message, Toast.LENGTH_LONG).show();
                }
                binding.tvBackupDetail.setText(failures > 0
                        ? "Hoàn tất một phần; còn " + failures + " media cần kiểm tra hoặc thử lại."
                        : "Lượt backup đã kết thúc.");
                setupUI();
                loadBackupStats();
            } else if (info.getState() == WorkInfo.State.FAILED) {
                binding.tvBackupDetail.setText("Backup thất bại. Kiểm tra mạng, quyền media và phiên đăng nhập rồi thử lại.");
            }
        } else if (periodic && periodicWorkWasRunning && info.getState() == WorkInfo.State.ENQUEUED) {
            periodicWorkWasRunning = false;
            binding.progressBar.setVisibility(View.GONE);
            binding.tvStatUploading.setText("0");
            loadBackupStats();
        }
    }

    private void loadCloudStorageUsage() {
        if (binding == null) return;
        UserInfo user = authRepository.getCurrentUser();
        if (user == null) {
            clearCloudStorageUsage();
            binding.tvCloudStorageStatus.setText(R.string.cloud_storage_refresh_failed);
            binding.progressCloudStorage.setVisibility(View.GONE);
            binding.btnRefreshCloudStorage.setEnabled(true);
            binding.tvCloudStorageUpdated.setVisibility(View.GONE);
            return;
        }
        String uid = user.getId();
        int generation = ++storageUsageGeneration;
        android.content.SharedPreferences cache = getSharedPreferences("cloud_storage_usage", MODE_PRIVATE);
        String bytesKey = "bytes:" + uid;
        String updatedKey = "updated:" + uid;
        boolean hasCache = cache.contains(bytesKey);
        final boolean hasValidCache;
        if (hasCache) {
            hasValidCache = renderCloudStorageUsage(cache.getLong(bytesKey, 0L), cache.getLong(updatedKey, 0L));
            binding.tvCloudStorageStatus.setText(hasValidCache
                    ? R.string.cloud_storage_refreshing : R.string.cloud_storage_refresh_failed);
        } else {
            hasValidCache = false;
            clearCloudStorageUsage();
            binding.tvCloudStorageUpdated.setVisibility(View.GONE);
            binding.tvCloudStorageStatus.setText(R.string.cloud_storage_loading);
        }
        binding.progressCloudStorage.setVisibility(View.VISIBLE);
        binding.btnRefreshCloudStorage.setEnabled(false);
        storageRepository.getCurrentUserStorageBytesAsync(uid, new StorageRepository.UsageCallback() {
            private boolean isCurrent() {
                UserInfo active = authRepository.getCurrentUser();
                return binding != null && !isFinishing() && !isDestroyed()
                        && generation == storageUsageGeneration && active != null && uid.equals(active.getId());
            }

            @Override public void onSuccess(long bytes) {
                if (!isCurrent()) return;
                if (!renderCloudStorageUsage(bytes, 0L)) {
                    onError("Invalid Storage usage");
                    return;
                }
                long updatedAt = System.currentTimeMillis();
                cache.edit().putLong(bytesKey, bytes).putLong(updatedKey, updatedAt).apply();
                renderCloudStorageUsage(bytes, updatedAt);
                binding.tvCloudStorageStatus.setText(R.string.cloud_storage_quota_note);
                binding.progressCloudStorage.setVisibility(View.GONE);
                binding.btnRefreshCloudStorage.setEnabled(true);
            }

            @Override public void onError(String error) {
                if (!isCurrent()) return;
                binding.progressCloudStorage.setVisibility(View.GONE);
                binding.btnRefreshCloudStorage.setEnabled(true);
                binding.tvCloudStorageStatus.setText(hasValidCache
                        ? R.string.cloud_storage_refresh_failed_cached
                        : R.string.cloud_storage_refresh_failed);
            }
        });
    }

    private boolean renderCloudStorageUsage(long bytes, long updatedAt) {
        if (binding == null) return false;
        final CloudStorageUsage usage;
        try {
            usage = CloudStorageUsage.fromBytes(bytes);
        } catch (IllegalArgumentException invalidUsage) {
            clearCloudStorageUsage();
            return false;
        }
        binding.tvCloudStorageUsed.setText(getString(R.string.cloud_storage_used_total, usage.usedAmount));
        binding.tvCloudStoragePercent.setText(getString(R.string.cloud_storage_percent_used, usage.percentText));
        binding.pbCloudStorageUsage.setProgress(usage.progress);
        binding.tvCloudStorageRemaining.setText(getString(R.string.cloud_storage_remaining, usage.remainingAmount));
        if (updatedAt > 0L) {
            String date = DateFormat.getDateTimeInstance(DateFormat.SHORT, DateFormat.SHORT)
                    .format(new Date(updatedAt));
            binding.tvCloudStorageUpdated.setText(getString(R.string.cloud_storage_updated, date));
            binding.tvCloudStorageUpdated.setVisibility(View.VISIBLE);
        } else {
            binding.tvCloudStorageUpdated.setVisibility(View.GONE);
        }
        return true;
    }

    private void clearCloudStorageUsage() {
        if (binding == null) return;
        binding.tvCloudStorageUsed.setText(R.string.cloud_storage_usage_unavailable);
        binding.tvCloudStoragePercent.setText(R.string.cloud_storage_usage_unavailable);
        binding.pbCloudStorageUsage.setProgress(0);
        binding.tvCloudStorageRemaining.setText(R.string.cloud_storage_usage_unavailable);
        binding.tvCloudStorageUpdated.setVisibility(View.GONE);
    }

    private void renderMediaTypeStats(int uploadedImagesThisRun, int uploadedVideosThisRun, String uploadingType) {
        if (latestStats == null) {
            binding.tvImageBackupStats.setText("Ảnh • Chưa quét");
            binding.tvVideoBackupStats.setText("Video • Chưa quét");
            return;
        }
        int uploadedImages = Math.min(latestStats.totalImages,
                latestStats.backedUpImages + uploadedImagesThisRun);
        int uploadedVideos = Math.min(latestStats.totalVideos,
                latestStats.backedUpVideos + uploadedVideosThisRun);
        int uploadedTotal = uploadedImages + uploadedVideos;
        int inFlight = uploadingType == null ? 0 : 1;
        binding.tvStatBackedUp.setText(String.valueOf(uploadedTotal));
        binding.tvStatRemaining.setText(String.valueOf(Math.max(0, latestStats.totalLocalPhotos - uploadedTotal - inFlight)));
        binding.pbBackupProgress.setProgress(latestStats.totalLocalPhotos == 0 ? 0
                : uploadedTotal * 100 / latestStats.totalLocalPhotos);
        int waitingImages = Math.max(0, latestStats.totalImages - uploadedImages - ("image".equals(uploadingType) ? 1 : 0));
        int waitingVideos = Math.max(0, latestStats.totalVideos - uploadedVideos - ("video".equals(uploadingType) ? 1 : 0));
        binding.tvImageBackupStats.setText("Ảnh • Tổng " + latestStats.totalImages
                + " · Đã sao lưu " + uploadedImages + " · Đang chờ "
                + waitingImages);
        binding.tvVideoBackupStats.setText("Video • Tổng " + latestStats.totalVideos
                + " · Đã sao lưu " + uploadedVideos + " · Đang chờ "
                + waitingVideos);
    }

    @Override protected void onResume() {
        super.onResume();
        loadCloudStorageUsage();
        if (prefs != null && hasMediaPermission()) {
            loadBackupStats();
            UserInfo user = authRepository.getCurrentUser();
            if (user != null && prefs.isBackupEnabled()) BackupWorkScheduler.reconcile(this, user.getId(), true);
        }
    }
}

        viewModel.getState().observe(this, state -> {
            if (state instanceof LoginViewModel.State.Loading) {
                setLoading(true);
            } else {
                setLoading(false);
            }

            if (state instanceof LoginViewModel.State.Success) {
                viewModel.consumeResult();
                Toast.makeText(this, "Đăng nhập thành công!", Toast.LENGTH_SHORT).show();

                intent.setFlags(Intent.FLAG_ACTIVITY_NEW_TASK | Intent.FLAG_ACTIVITY_CLEAR_TASK);
                startActivity(intent);
                finish();
            } else if (state instanceof LoginViewModel.State.Error) {
                String message = ((LoginViewModel.State.Error) state).getMessage();
                viewModel.consumeResult();
                Toast.makeText(this, message, Toast.LENGTH_LONG).show();
            }
        });

        String sessionError = getIntent().getStringExtra("auth_session_error");
        if (!TextUtils.isEmpty(sessionError)) {
            Toast.makeText(this, sessionError, Toast.LENGTH_LONG).show();
            getIntent().removeExtra("auth_session_error");
        }

        binding.btnLogin.setOnClickListener(v -> performLogin());

        binding.tvForgotPassword.setOnClickListener(v -> {
            startActivity(new Intent(LoginActivity.this, ForgotPasswordActivity.class));
        });

        binding.tvGoToRegister.setOnClickListener(v -> {
            startActivity(new Intent(LoginActivity.this, RegisterActivity.class));
        });
    }

    private void performLogin() {
        String email = binding.etEmail.getText() != null ? binding.etEmail.getText().toString().trim() : "";
        String password = binding.etPassword.getText() != null ? binding.etPassword.getText().toString() : "";

        if (TextUtils.isEmpty(email)) {
            binding.tilEmail.setError("Vui lòng nhập Email");
            return;
        } else {
            binding.tilEmail.setError(null);
        }

        if (!Patterns.EMAIL_ADDRESS.matcher(email).matches()) {
            binding.tilEmail.setError("Nhập địa chỉ email hợp lệ");
            return;
        }

        if (TextUtils.isEmpty(password)) {
            binding.tilPassword.setError("Vui lòng nhập mật khẩu");
            return;
        } else {
            binding.tilPassword.setError(null);
        }

        viewModel.login(email, password);
    }

    private void setLoading(boolean isLoading) {
        binding.progressBar.setVisibility(isLoading ? View.VISIBLE : View.GONE);
        binding.btnLogin.setEnabled(!isLoading);
    }

package com.example.th6.data

import io.github.jan.supabase.auth.auth
import io.github.jan.supabase.auth.exception.AuthRestException
import io.github.jan.supabase.auth.exception.AuthErrorCode
import io.github.jan.supabase.auth.providers.builtin.Email

import io.ktor.client.plugins.HttpRequestTimeoutException
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext
import java.io.IOException
import java.net.UnknownHostException

class AuthRepository {

    interface AuthCallback {
        fun onSuccess(userId: String?, message: String)
        fun onError(error: String)
    }

    interface SessionCallback {
        fun onResult(isLoggedIn: Boolean, userId: String?, errorMessage: String?)
    }

    interface LogoutCallback {
        fun onComplete(result: LogoutResult)
    }

    data class LogoutResult(
        val localSessionCleared: Boolean,
        val serverSignOutConfirmed: Boolean
    )

    sealed class AuthResult {
        data class Success(val message: String, val userId: String? = null) : AuthResult()
        data class Failure(val exception: Throwable, val message: String) : AuthResult()
    }

    private val auth by lazy {
        SupabaseManager.client.auth
    }

    /**
     * Signs up a new user using email and password via Supabase Auth.
     */
    suspend fun signUp(email: String, pass: String): AuthResult {
        return withContext(Dispatchers.IO) {
            try {
                auth.signUpWith(Email) {
                    this.email = email
                    this.password = pass
                }
                val user = auth.currentUserOrNull()
                AuthResult.Success("Đăng ký thành công!", user?.id)
            } catch (e: Exception) {
                AuthResult.Failure(e, e.localizedMessage ?: "Đăng ký tài khoản thất bại")
            }
        }
    }

    /**
     * Signs in an existing user using email and password via Supabase Auth.
     */
    suspend fun signIn(email: String, pass: String): AuthResult {
        return withContext(Dispatchers.IO) {
            try {
                auth.signInWith(Email) {
                    this.email = email
                    this.password = pass
                }
                val session = auth.currentSessionOrNull()
                val userId = session?.user?.id
                if (userId.isNullOrBlank()) {
                    AuthResult.Failure(IllegalStateException("Supabase returned no authenticated session"),
                        "Không thể xác nhận phiên đăng nhập. Vui lòng thử lại.")
                } else {
                    AuthResult.Success("Đăng nhập thành công!", userId)
                }
            } catch (e: Exception) {
                AuthResult.Failure(e, readableAuthError(e))
            }
        }
    }

    private fun readableAuthError(error: Throwable): String {
        var current: Throwable? = error
        while (current != null) {
            when (current) {
                is AuthRestException -> {
                    when (current.errorCode) {
                        AuthErrorCode.InvalidCredentials, AuthErrorCode.UserNotFound ->
                            return "Email hoặc mật khẩu không đúng."
                        AuthErrorCode.EmailNotConfirmed ->
                            return "Email của bạn chưa được xác nhận. Hãy kiểm tra hộp thư."
                        AuthErrorCode.OverRequestRateLimit ->
                            return "Bạn thử đăng nhập quá nhiều lần. Vui lòng đợi rồi thử lại."
                        else -> if (current.statusCode >= 500) {
                            return "Máy chủ xác thực đang gặp sự cố. Vui lòng thử lại sau."
                        }
                    }
                }
                is HttpRequestException, is HttpRequestTimeoutException,
                is UnknownHostException, is IOException ->
                    return "Không thể kết nối máy chủ. Kiểm tra mạng rồi thử lại."
                is RestException -> if (current.statusCode >= 500) {
                    return "Máy chủ xác thực đang gặp sự cố. Vui lòng thử lại sau."
                }
            }
            current = current.cause
        }
        return "Đăng nhập chưa thành công. Vui lòng kiểm tra thông tin và thử lại."
    }

    /**
     * Sends password reset email.
     */
    suspend fun resetPassword(email: String): AuthResult {
        return withContext(Dispatchers.IO) {
            try {
                auth.resetPasswordForEmail(email, redirectUrl = SupabaseManager.AUTH_REDIRECT_URL)
                AuthResult.Success("Đã gửi email hướng dẫn đặt lại mật khẩu.")
            } catch (e: Exception) {
                AuthResult.Failure(e, recoveryRequestError(e))
            }
        }
    }

    suspend fun acceptRecoveryFragment(fragment: String): AuthResult = withContext(Dispatchers.IO) {
        try {
            val session = auth.parseSessionFromFragment(fragment)
            if (!session.type.equals("recovery", ignoreCase = true)) {
                return@withContext AuthResult.Failure(IllegalStateException(), "Liên kết khôi phục không hợp lệ hoặc đã hết hạn.")
            }
            // Verify with Supabase before persisting the external callback session.
            val user = auth.retrieveUser(session.accessToken)
            auth.importSession(session.copy(user = user), source = SessionSource.External)
            AuthResult.Success("recovery")
        } catch (e: Exception) {
            AuthResult.Failure(e, "Liên kết khôi phục không hợp lệ hoặc đã hết hạn. Hãy yêu cầu liên kết mới.")
        }
    }

    suspend fun updateRecoveredPassword(password: String): AuthResult = withContext(Dispatchers.IO) {
        try {
            val session = auth.currentSessionOrNull()
            if (session == null || !session.type.equals("recovery", ignoreCase = true)) {
                return@withContext AuthResult.Failure(IllegalStateException(), "Phiên khôi phục không hợp lệ hoặc đã hết hạn. Hãy yêu cầu liên kết mới.")
            }
            auth.updateUser { this.password = password }
            auth.clearSession()
            AuthResult.Success("Mật khẩu đã được cập nhật. Vui lòng đăng nhập lại.")
        } catch (e: Exception) {
            AuthResult.Failure(e, readableRecoveryUpdateError(e))
        }
    }

    private fun recoveryRequestError(error: Throwable): String = when {
        isNetworkError(error) -> "Không thể kết nối. Kiểm tra mạng rồi thử lại."
        error is AuthRestException && error.statusCode >= 500 -> "Máy chủ xác thực đang gặp sự cố. Vui lòng thử lại sau."
        error is RestException && error.statusCode >= 500 -> "Máy chủ xác thực đang gặp sự cố. Vui lòng thử lại sau."
        else -> "Chưa thể gửi yêu cầu khôi phục. Vui lòng thử lại sau."
    }

    private fun readableRecoveryUpdateError(error: Throwable): String {
        var current: Throwable? = error
        while (current != null) {
            if (current is AuthRestException && current.errorCode == AuthErrorCode.WeakPassword) {
                return "Mật khẩu chưa đáp ứng chính sách bảo mật. Hãy chọn mật khẩu khác."
            }
            if (current is RestException && current.statusCode >= 500 || current is AuthRestException && current.statusCode >= 500) {
                return "Máy chủ xác thực đang gặp sự cố. Vui lòng thử lại sau."
            }
            if (isNetworkError(current)) return "Không thể kết nối. Kiểm tra mạng rồi thử lại."
            current = current.cause
        }
        return "Chưa thể cập nhật mật khẩu. Phiên có thể đã hết hạn; hãy yêu cầu liên kết mới."
    }

    private fun isNetworkError(error: Throwable): Boolean {
        var current: Throwable? = error
        while (current != null) {
            if (current is HttpRequestException || current is HttpRequestTimeoutException ||
                current is UnknownHostException || current is IOException) return true
            current = current.cause
        }
        return false
    }

    /**
     * Checks if the current user has a valid active session in Supabase Auth.
     */
    suspend fun checkIsLoggedIn(): Boolean {
        return withContext(Dispatchers.IO) {
            try {
                auth.awaitInitialization()
                auth.currentSessionOrNull()?.user?.id?.isNotBlank() == true
            } catch (e: Exception) {
                false
            }
        }
    }

    /**
     * Synchronous session check returning true if current user info is cached.
     */
    fun isLoggedIn(): Boolean {
        return auth.currentUserOrNull() != null && auth.currentSessionOrNull() != null
    }

    /**
     * Signs out the user session from Supabase Auth and clears session tokens.
     */
    suspend fun signOut(): LogoutResult {
        return withContext(Dispatchers.IO) {
            try {
                auth.signOut()
                val cleared = auth.currentSessionOrNull() == null
                if (cleared) LogoutResult(true, true) else {
                    auth.clearSession()
                    LogoutResult(auth.currentSessionOrNull() == null, false)
                }
            } catch (e: Exception) {
                // A network failure may prevent remote token revocation. Always clear the
                // device session so the user cannot continue into private screens here.
                try {
                    auth.clearSession()
                    LogoutResult(auth.currentSessionOrNull() == null, false)
                } catch (_: Exception) {
                    LogoutResult(false, false)
                }
            }
        }
    }

    /**
     * Returns the currently authenticated user info, or null if not logged in.
     */
    fun getCurrentUser(): UserInfo? {
        return auth.currentUserOrNull()
    }

    /**
     * Returns the currently authenticated user's ID, or null.
     */
    fun getCurrentUserUid(): String? {
        return auth.currentUserOrNull()?.id
    }

    // --- Java Async Callback Helpers ---

    fun checkIsLoggedInAsync(callback: SessionCallback) {
        CoroutineScope(Dispatchers.IO).launch {
            var loggedIn = false
            var uid: String? = null
            var errorMessage: String? = null
            try {
                auth.awaitInitialization()
                when (val status = auth.sessionStatus.value) {
                    is SessionStatus.Authenticated -> {
                        uid = status.session.user?.id
                        loggedIn = !uid.isNullOrBlank()
                    }
                    is SessionStatus.RefreshFailure -> {
                        errorMessage = "Không thể xác minh phiên đăng nhập. Kiểm tra mạng rồi thử lại."
                    }
                    else -> Unit
                }
            } catch (_: Exception) {
                errorMessage = "Không thể kiểm tra phiên đăng nhập. Kiểm tra mạng rồi thử lại."
            }
            withContext(Dispatchers.Main) {
                callback.onResult(loggedIn, uid, errorMessage)
            }
        }
    }

    fun signUpAsync(email: String, pass: String, callback: AuthCallback) {
        CoroutineScope(Dispatchers.IO).launch {
            when (val result = signUp(email, pass)) {
                is AuthResult.Success -> {
                    withContext(Dispatchers.Main) {
                        callback.onSuccess(result.userId, result.message)
                    }
                }
                is AuthResult.Failure -> {
                    withContext(Dispatchers.Main) {
                        callback.onError(result.message)
                    }
                }
            }
        }
    }

    fun signInAsync(email: String, pass: String, callback: AuthCallback) {
        CoroutineScope(Dispatchers.IO).launch {
            when (val result = signIn(email, pass)) {
                is AuthResult.Success -> {
                    withContext(Dispatchers.Main) {
                        callback.onSuccess(result.userId, result.message)
                    }
                }
                is AuthResult.Failure -> {
                    withContext(Dispatchers.Main) {
                        callback.onError(result.message)
                    }
                }
            }
        }
    }

    fun resetPasswordAsync(email: String, callback: AuthCallback) {
        CoroutineScope(Dispatchers.IO).launch {
            when (val result = resetPassword(email)) {
                is AuthResult.Success -> {
                    withContext(Dispatchers.Main) {
                        callback.onSuccess(null, result.message)
                    }
                }
                is AuthResult.Failure -> {
                    withContext(Dispatchers.Main) {
                        callback.onError(result.message)
                    }
                }
            }
        }
    }

    fun signOutAsync(callback: LogoutCallback?) {
        CoroutineScope(Dispatchers.IO).launch {
            val result = signOut()
            callback?.let {
                withContext(Dispatchers.Main) {
                    it.onComplete(result)
                }
            }
        }
    }
}

class PhotoRepository {

    interface Callback<T> {
        fun onSuccess(result: T)
        fun onError(error: String)
    }

    companion object {
        const val TABLE_PHOTOS = "photos"
        const val TABLE_ALBUMS = "albums"
        const val TABLE_USERS = "users"
    }

    private val photosTable by lazy { SupabaseManager.client.from(TABLE_PHOTOS) }
    private val albumsTable by lazy { SupabaseManager.client.from(TABLE_ALBUMS) }
    private val usersTable by lazy { SupabaseManager.client.from(TABLE_USERS) }

    // --- PHOTOS ---

    suspend fun getAllPhotos(): List<Photo> {
        return withContext(Dispatchers.IO) {
            photosTable.select().decodeList<Photo>()
        }
    }

    suspend fun getPhotosByUser(userId: String): List<Photo> {
        return withContext(Dispatchers.IO) {
            photosTable.select {
                filter {
                    eq("user_id", userId)
                }
            }.decodeList<Photo>()
        }
    }

    /** Fetch the complete account library in bounded PostgREST pages for media analysis. */
    suspend fun getPhotosByUserPaged(userId: String, pageSize: Long = 500L): List<Photo> = withContext(Dispatchers.IO) {
        require(userId.isNotBlank()) { "A signed-in UID is required" }
        require(pageSize in 1L..1000L) { "Page size must be between 1 and 1000" }
        val result = ArrayList<Photo>()
        var offset = 0L
        while (true) {
            val page = photosTable.select {
                filter { eq("user_id", userId) }
                order("id", Order.ASCENDING)
                range(offset, offset + pageSize - 1)
            }.decodeList<Photo>()
            result.addAll(page)
            if (page.size.toLong() < pageSize) break
            offset += pageSize
        }
        result
    }

    suspend fun getPhotoByUser(userId: String, photoId: String): Photo? = withContext(Dispatchers.IO) {
        photosTable.select {
            filter { eq("user_id", userId); eq("id", photoId) }
            range(0, 0)
        }.decodeList<Photo>().firstOrNull()
    }

    suspend fun getPhotos(userId: String): List<Photo> = getPhotosByUser(userId)

    suspend fun insertPhoto(photo: Photo) {
        withContext(Dispatchers.IO) {
            photosTable.insert(photo)
        }
    }

    suspend fun savePhoto(photo: Photo) {
        withContext(Dispatchers.IO) {
            photosTable.upsert(photo)
        }
    }

    suspend fun saveBackupPhoto(photo: Photo) {
        val uri = photo.localMediaUri
        if (photo.id.isEmpty() && !uri.isNullOrEmpty()) {
            // Include the account in the deterministic key: the same MediaStore URI
            // can exist for different signed-in users on one device.
            photo.id = "backup_" + sha256(photo.ownerUid + "|" + uri)
        } else if (photo.id.isEmpty()) {
            photo.id = "backup_" + sha256(photo.ownerUid + "|" + System.currentTimeMillis())
        }
        withContext(Dispatchers.IO) {
            photosTable.upsert(photo)
        }
    }

    private fun sha256(value: String): String {
        try {
            val digest = MessageDigest.getInstance("SHA-256").digest(value.toByteArray(StandardCharsets.UTF_8))
            val result = StringBuilder(digest.size * 2)
            for (b in digest) {
                result.append(String.format(java.util.Locale.US, "%02x", b.toInt() and 0xff))
            }
            return result.toString()
        } catch (e: NoSuchAlgorithmException) {
            throw IllegalStateException("SHA-256 unavailable", e)
        }
    }

    suspend fun updatePhotoAlbums(userId: String, photoId: String, albumIds: List<String>) {
        withContext(Dispatchers.IO) {
            val ownedAlbumIds = getAlbums(userId).map { it.id }.toSet()
            val current = getPhotosByUser(userId).firstOrNull { it.id == photoId }
                ?: throw IllegalStateException("Photo row was not found for the current user")
            val newlyRequestedForeignIds = albumIds.filter { it !in ownedAlbumIds && it !in current.albumIds }
            if (newlyRequestedForeignIds.isNotEmpty()) {
                throw IllegalStateException("Album membership contains an album not owned by the current user")
            }
            // Drop stale IDs left by an album deleted earlier, while retaining every
            // currently owned relationship supplied by the caller.
            val normalizedIds = albumIds.distinct().filter { it in ownedAlbumIds }
            photosTable.update(
                buildJsonObject { put("album_ids", kotlinx.serialization.json.JsonArray(normalizedIds.map(::JsonPrimitive))) }
            ) {
                filter { eq("id", photoId); eq("user_id", userId) }
            }
            val updated = getPhotosByUser(userId).firstOrNull { it.id == photoId }
                ?: throw IllegalStateException("Photo row was not found for the current user")
            if (updated.albumIds.toSet() != normalizedIds.toSet()) throw IllegalStateException("Album membership was not updated")
        }
    }

    suspend fun addPhotoToAlbum(userId: String, photoId: String, albumId: String) {
        val ownedAlbumIds = getAlbums(userId).map { it.id }.toSet()
        if (albumId !in ownedAlbumIds) throw IllegalStateException("Album was not found for the current user")
        val photo = getPhotosByUser(userId).firstOrNull { it.id == photoId }
            ?: throw IllegalStateException("Photo row was not found for the current user")
        val ids = photo.albumIds.filter { it in ownedAlbumIds }.toMutableList()
        if (!ids.contains(albumId)) ids.add(albumId)
        updatePhotoAlbums(userId, photoId, ids)
    }

    suspend fun removePhotoFromAlbum(userId: String, photoId: String, albumId: String) {
        val ownedAlbumIds = getAlbums(userId).map { it.id }.toSet()
        if (albumId !in ownedAlbumIds) throw IllegalStateException("Album was not found for the current user")
        val photo = getPhotosByUser(userId).firstOrNull { it.id == photoId }
            ?: throw IllegalStateException("Photo row was not found for the current user")
        updatePhotoAlbums(userId, photoId, photo.albumIds.filter { it != albumId && it in ownedAlbumIds })
    }

    suspend fun updateUserProfile(
        uid: String,
        updates: Map<String, Any>,
        fallbackDisplayName: String,
        fallbackEmail: String
    ) {
        withContext(Dispatchers.IO) {
            require(uid.isNotBlank()) { "A signed-in user is required" }
            require(updates.keys.all { it == "display_name" || it == "photo_url" }) {
                "Unsupported profile field"
            }

            // Email confirmation can leave a valid Auth user without a row in public.users.
            // A PostgREST UPDATE against that missing row succeeds with zero rows, so merge
            // the requested fields into a complete profile and upsert it instead.
            val existing = usersTable.select {
                filter { eq("uid", uid) }
            }.decodeSingleOrNull<UserProfile>()

            val profile = UserProfile(
                uid = uid,
                displayName = (updates["display_name"] as? String)
                    ?: existing?.displayName?.takeIf { it.isNotBlank() }
                    ?: fallbackDisplayName,
                email = existing?.email?.takeIf { it.isNotBlank() } ?: fallbackEmail,
                photoUrl = (updates["photo_url"] as? String)
                    ?: existing?.photoUrl.orEmpty(),
                createdAt = existing?.createdAt?.takeIf { it > 0L } ?: System.currentTimeMillis()
            )

            usersTable.upsert(profile)

            val saved = usersTable.select {
                filter { eq("uid", uid) }
            }.decodeSingleOrNull<UserProfile>()
                ?: throw IllegalStateException("Profile row was not saved")
            if (saved.displayName != profile.displayName || saved.photoUrl != profile.photoUrl) {
                throw IllegalStateException("Profile values did not persist")
            }
        }
    }

    private fun Map<String, Any>.toJsonObject(): JsonObject = buildJsonObject {
        for ((key, value) in this@toJsonObject) {
            when (value) {
                is String -> put(key, JsonPrimitive(value))
                is Boolean -> put(key, JsonPrimitive(value))
                is Int -> put(key, JsonPrimitive(value))
                is Long -> put(key, JsonPrimitive(value))
                is Double -> put(key, JsonPrimitive(value))
                is Float -> put(key, JsonPrimitive(value.toDouble()))
                else -> throw IllegalArgumentException("Unsupported update value type")
            }
        }
    }

package com.example.th6.data

import android.content.Context
import android.net.Uri
import android.provider.OpenableColumns
import io.github.jan.supabase.storage.storage
import io.github.jan.supabase.storage.upload
import io.github.jan.supabase.storage.FileObject
import io.ktor.http.ContentType
import kotlinx.coroutines.CoroutineScope

import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext

/** Storage operations for the public PhotoVault bucket. URI uploads use SDK streaming APIs. */
class StorageRepository {
    interface UploadCallback {
        fun onSuccess(publicUrl: String)
        fun onError(error: String)
    }
    interface DeleteCallback {
        fun onSuccess()
        fun onError(error: String)
    }
    interface UsageCallback {
        fun onSuccess(bytes: Long)
        fun onError(error: String)
    }
    sealed class Result {
        data class Success(val publicUrl: String) : Result()
        data class Failure(val exception: Throwable, val message: String) : Result()
    }
    companion object {
        const val BUCKET_NAME = "photos_bucket"
        const val MAX_FILE_BYTES = 50L * 1024L * 1024L
    }

    private val bucket by lazy { SupabaseManager.client.storage.from(BUCKET_NAME) }
    fun getImageUrl(path: String): String = bucket.publicUrl(path)

    fun objectPathFromPublicUrl(publicUrl: String?): String {
        if (publicUrl.isNullOrBlank()) return ""
        val segments = Uri.parse(publicUrl).pathSegments
        val index = segments.indexOf(BUCKET_NAME)
        return if (index >= 0 && index + 1 < segments.size) segments.subList(index + 1, segments.size).joinToString("/") else ""
    }

    /** Upload a content URI without buffering the entire image/video in memory. */
    suspend fun uploadMedia(context: Context, uri: Uri, path: String, mimeType: String?): Result = withContext(Dispatchers.IO) {
        try {
            val appContext = context.applicationContext
            val size = querySize(appContext, uri)
            if (size <= 0L) return@withContext Result.Failure(IllegalArgumentException("Unknown media size"), "Không xác định được dung lượng tệp.")
            if (size > MAX_FILE_BYTES) return@withContext Result.Failure(IllegalArgumentException("Media exceeds bucket limit"), "Tệp vượt giới hạn 50 MiB của bucket.")
            appContext.contentResolver.openAssetFileDescriptor(uri, "r")?.use { }
                ?: return@withContext Result.Failure(IllegalArgumentException("Unreadable URI"), "Không còn quyền đọc tệp đã chọn.")
            bucket.upload(path = path, uri = uri) {
                upsert = true
                contentType = mimeType?.takeIf { it.contains('/') }?.let { ContentType.parse(it) }
            }
            Result.Success(bucket.publicUrl(path))
        } catch (e: Exception) {
            val message = when {
                e is SecurityException -> "Không còn quyền đọc tệp. Hãy chọn lại tệp và thử lại."
                e.cause is java.io.IOException || e is java.io.IOException -> "Không thể đọc tệp hoặc kết nối mạng. Kiểm tra quyền truy cập và mạng rồi thử lại."
                else -> e.localizedMessage?.takeIf { it.isNotBlank() } ?: "Không thể tải tệp lên Supabase Storage."
            }
            Result.Failure(e, message)
        }
    }

    private fun querySize(context: Context, uri: Uri): Long {
        context.contentResolver.query(uri, arrayOf(OpenableColumns.SIZE), null, null, null)?.use { cursor ->
            if (cursor.moveToFirst()) {
                val index = cursor.getColumnIndex(OpenableColumns.SIZE)
                if (index >= 0 && !cursor.isNull(index)) return cursor.getLong(index)
            }
        }
        return context.contentResolver.openAssetFileDescriptor(uri, "r")?.use { it.length } ?: -1L
    }

    suspend fun uploadImage(byteArray: ByteArray, fileName: String): Result = withContext(Dispatchers.IO) {
        try {
            if (byteArray.size > MAX_FILE_BYTES) return@withContext Result.Failure(IllegalArgumentException("Image exceeds limit"), "Tệp vượt giới hạn 50 MiB.")
            bucket.upload(path = fileName, data = byteArray) { upsert = true }
            Result.Success(bucket.publicUrl(fileName))
        } catch (e: Exception) { Result.Failure(e, e.localizedMessage ?: "Lỗi upload ảnh lên Supabase Storage") }
    }

    suspend fun uploadImage(context: Context, uri: Uri, fileName: String): Result = uploadMedia(context, uri, fileName, context.contentResolver.getType(uri))

    suspend fun deleteImage(path: String): Result = withContext(Dispatchers.IO) {
        try { bucket.delete(path); Result.Success(bucket.publicUrl(path)) }
        catch (e: Exception) { Result.Failure(e, e.localizedMessage ?: "Lỗi xóa tệp khỏi Supabase Storage") }
    }

    /** Sum the actual object sizes visible to the signed-in user through Storage RLS. */
    suspend fun getCurrentUserStorageBytes(userId: String): Long = withContext(Dispatchers.IO) {
        require(userId.isNotBlank()) { "A signed-in user is required" }
        val pendingPrefixes = ArrayDeque<String>()
        pendingPrefixes.add("")
        var total = 0L
        while (pendingPrefixes.isNotEmpty()) {
            val prefix = pendingPrefixes.removeFirst()
            var offset = 0
            while (true) {
                val page = bucket.list(prefix = prefix) {
                    limit = 100
                    this.offset = offset
                }
                for (item: FileObject in page) {
                    if (item.id == null) {
                        pendingPrefixes.add(prefix + item.name.trimEnd('/') + "/")
                    } else {
                        val rawSize = item.metadata?.get("size")?.toString()?.trim('"')
                        val size = rawSize?.toLongOrNull()
                            ?: throw IllegalStateException("Storage did not return a valid size for ${item.name}")
                        if (size < 0L) throw IllegalStateException("Storage returned a negative object size")
                        total = Math.addExact(total, size)
                    }
                }
                if (page.size < 100) break
                offset += page.size
            }
        }
        total
    }

    fun getCurrentUserStorageBytesAsync(userId: String, callback: UsageCallback) = CoroutineScope(Dispatchers.IO).launch {
        try {
            val bytes = getCurrentUserStorageBytes(userId)
            withContext(Dispatchers.Main) { callback.onSuccess(bytes) }
        } catch (e: Exception) {
            withContext(Dispatchers.Main) { callback.onError(e.localizedMessage ?: "Không thể tải dung lượng Storage") }
        }
    }

    fun uploadImageAsync(byteArray: ByteArray, fileName: String, callback: UploadCallback) = CoroutineScope(Dispatchers.IO).launch {
        dispatchUpload(uploadImage(byteArray, fileName), callback)
    }
    fun uploadImageAsync(context: Context, uri: Uri, fileName: String, callback: UploadCallback) = CoroutineScope(Dispatchers.IO).launch {
        dispatchUpload(uploadImage(context, uri, fileName), callback)
    }
    fun uploadMediaAsync(context: Context, uri: Uri, path: String, mimeType: String?, callback: UploadCallback) = CoroutineScope(Dispatchers.IO).launch {
        dispatchUpload(uploadMedia(context, uri, path, mimeType), callback)
    }
    private suspend fun dispatchUpload(result: Result, callback: UploadCallback) {
        withContext(Dispatchers.Main) {
            when (result) {
                is Result.Success -> callback.onSuccess(result.publicUrl)
                is Result.Failure -> callback.onError(result.message)
            }
        }
    }
    fun deleteImageAsync(path: String, callback: DeleteCallback) = CoroutineScope(Dispatchers.IO).launch {
        when (val result = deleteImage(path)) {
            is Result.Success -> withContext(Dispatchers.Main) { callback.onSuccess() }
            is Result.Failure -> withContext(Dispatchers.Main) { callback.onError(result.message) }
        }
    }
}

package com.example.th6.viewmodels

class PhotoDeleteViewModel(
    private val storageRepository: StorageRepository = StorageRepository(),
    private val photoRepository: PhotoRepository = PhotoRepository()
) : ViewModel() {

    sealed class DeleteState {
        object Idle : DeleteState()
        data class Progress(val message: String = "Đang xóa ảnh khỏi Supabase...") : DeleteState()
        data class Success(val photoId: String, val message: String = "Xóa ảnh thành công!") : DeleteState()
        data class Error(val error: String) : DeleteState()
    }

    private val _deleteState = MutableLiveData<DeleteState>(DeleteState.Idle)
    val deleteState: LiveData<DeleteState> get() = _deleteState

    interface DeleteCallback {
        fun onSuccess(photoId: String)
        fun onError(error: String)
    }

    /**
     * Executes photo deletion flow on Supabase:
     * (1) Calls StorageRepository.deleteImage to delete the physical file on Supabase Storage ('photos_bucket').
     * (2) Calls PhotoRepository.deletePhoto to delete the metadata record in Supabase 'photos' table.
     * Completely removes all legacy Firebase logic.
     */
    fun deletePhoto(photo: Photo) {
        viewModelScope.launch {
            _deleteState.value = DeleteState.Progress("Đang xóa tệp và dữ liệu khỏi Supabase...")

            val photoId = photo.id
            if (photoId.isBlank()) {
                _deleteState.value = DeleteState.Error("ID ảnh không hợp lệ.")
                return@launch
            }

            val fileName = photo.storagePath.takeIf { it.isNotBlank() }
                ?: storageRepository.objectPathFromPublicUrl(photo.imageUrl)

            var storageError: String? = null
            var databaseError: String? = null

            // 1. Delete physical file from Supabase Storage ('photos_bucket')
            if (fileName.isNotBlank()) {
                val storageResult = storageRepository.deleteImage(fileName)
                if (storageResult is StorageRepository.Result.Failure) {
                    storageError = storageResult.message
                }
            }

            // 2. Delete metadata record from Supabase 'photos' database table
            try {
                photoRepository.deletePhoto(photoId)
            } catch (e: Exception) {
                databaseError = e.localizedMessage ?: "Lỗi xóa dữ liệu trên Database"
            }

            // Report result to UI
            if (databaseError != null) {
                _deleteState.value = DeleteState.Error("Lỗi xóa cơ sở dữ liệu: $databaseError")
            } else if (storageError != null) {
                _deleteState.value = DeleteState.Success(photoId, "Đã xóa bản ghi dữ liệu (Lưu ý Storage: $storageError)")
            } else {
                _deleteState.value = DeleteState.Success(photoId, "Đã xóa hoàn toàn tệp và dữ liệu ảnh khỏi Supabase!")
            }
        }
    }

    /**
     * Overload accepting photoId and fileName directly.
     */
    fun deletePhoto(photoId: String, fileName: String) {
        viewModelScope.launch {
            _deleteState.value = DeleteState.Progress("Đang xóa ảnh khỏi Supabase...")

            // 1. Delete physical file on Supabase Storage
            val storageResult = storageRepository.deleteImage(fileName)

            // 2. Delete record from Supabase 'photos' database table
            try {
                photoRepository.deletePhoto(photoId)
                if (storageResult is StorageRepository.Result.Failure) {
                    _deleteState.value = DeleteState.Success(photoId, "Đã xóa bản ghi cơ sở dữ liệu.")
                } else {
                    _deleteState.value = DeleteState.Success(photoId, "Xóa ảnh khỏi Supabase thành công!")
                }
            } catch (e: Exception) {
                _deleteState.value = DeleteState.Error("Lỗi xóa bản ghi Database: ${e.localizedMessage}")
            }
        }
    }

    // --- Java Async Callback Helpers ---

    fun deletePhotoAsync(photo: Photo, callback: DeleteCallback) {
        viewModelScope.launch {
            deletePhoto(photo)
            when (val state = _deleteState.value) {
                is DeleteState.Success -> {
                    withContext(Dispatchers.Main) {
                        callback.onSuccess(state.photoId)
                    }
                }
                is DeleteState.Error -> {
                    withContext(Dispatchers.Main) {
                        callback.onError(state.error)
                    }
                }
                else -> {}
            }
        }
    }
}

package com.example.th6.services;

import android.content.Context;
import android.content.pm.PackageManager;
import android.Manifest;
import android.net.Uri;
import android.os.Build;
import android.os.SystemClock;

import androidx.annotation.NonNull;
import androidx.core.content.ContextCompat;
import androidx.work.Data;
import androidx.work.Worker;
import androidx.work.WorkerParameters;

import com.example.th6.data.AuthRepository;
import com.example.th6.data.PhotoRepository;
import com.example.th6.data.StorageRepository;
import com.example.th6.models.LocalMediaPhoto;
import com.example.th6.models.Photo;
import com.example.th6.utils.BackupPreferences;
import io.github.jan.supabase.auth.user.UserInfo;

import java.util.List;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.atomic.AtomicReference;
import java.util.concurrent.locks.ReentrantLock;

public class PhotoBackupWorker extends Worker {
    private static final long CALLBACK_TIMEOUT_SECONDS = 210;
    private static final long MAX_RUN_MILLIS = TimeUnit.MINUTES.toMillis(8);
    private static final ReentrantLock RUN_LOCK = new ReentrantLock();

    public PhotoBackupWorker(@NonNull Context context, @NonNull WorkerParameters workerParams) {
        super(context, workerParams);
    }

    static boolean acquireIdleRunLock() { return RUN_LOCK.tryLock(); }
    static void releaseIdleRunLock() { RUN_LOCK.unlock(); }

    @NonNull @Override public Result doWork() {
        if (!RUN_LOCK.tryLock()) return Result.retry();
        try {
            return runBackup();
        } finally {
            RUN_LOCK.unlock();
        }
    }

    @NonNull private Result runBackup() {
        long runStartedAt = SystemClock.elapsedRealtime();
        Context context = getApplicationContext();
        String requestedUid = getInputData().getString("backup_uid");
        AuthRepository authRepository = new AuthRepository();
        UserInfo user = authRepository.isLoggedIn() ? authRepository.getCurrentUser() : null;
        if (user == null) {
            if (requestedUid != null) BackupWorkScheduler.cancelPeriodic(context, requestedUid);
            return Result.success(output(0, 0, 0, "Bạn cần đăng nhập để backup."));
        }
        String uid = user.getId();
        if (requestedUid != null && !requestedUid.equals(uid)) {
            BackupWorkScheduler.cancelPeriodic(context, requestedUid);
            return Result.success(output(0, 0, 0, "Tài khoản đã thay đổi; bỏ qua lượt backup cũ."));
        }
        BackupPreferences prefs = new BackupPreferences(context, uid);
        if (!prefs.isBackupEnabled() || prefs.isPaused()) {
            BackupWorkScheduler.cancelPeriodic(context, uid);
            return Result.success(output(0, 0, 0, "Backup đang tắt hoặc tạm dừng."));
        }
        if (!hasAnyMediaReadPermission(context)) {
            return Result.success(output(0, 0, 0, "Chưa có quyền đọc ảnh; backup sẽ thử lại sau khi được cấp quyền."));
        }

        AtomicReference<List<LocalMediaPhoto>> pendingRef = new AtomicReference<>();
        AtomicReference<PhotoBackupManager.BackupStats> statsRef = new AtomicReference<>();
        AtomicReference<String> scanError = new AtomicReference<>();
        CountDownLatch scanLatch = new CountDownLatch(1);
        new PhotoBackupManager().processBackupState(context, uid, new PhotoBackupManager.ManagerCallback() {
            @Override public void onStatsReady(PhotoBackupManager.BackupStats stats, List<LocalMediaPhoto> all) {
                statsRef.set(stats);
                pendingRef.set(stats.pendingPhotos);
                scanLatch.countDown();
            }
            @Override public void onError(String error) { scanError.set(error); scanLatch.countDown(); }
        });
        if (!await(scanLatch, 45)) return Result.retry();
        if (scanError.get() != null) return Result.retry();
        List<LocalMediaPhoto> pending = pendingRef.get();
        PhotoBackupManager.BackupStats initialStats = statsRef.get();
        if (!prefs.isBackupEnabled() || prefs.isPaused()) {
            BackupWorkScheduler.cancelPeriodic(context, uid);
            return Result.success(output(0, 0, pending == null ? 0 : pending.size(), "Backup đang tắt hoặc tạm dừng."));
        }
        if (pending == null || pending.isEmpty()) {
            prefs.setLastSyncTime(System.currentTimeMillis());
            return Result.success(output(0, 0, 0, "Không có ảnh cần backup."));
        }

        StorageRepository storageRepository = new StorageRepository();
        PhotoRepository photoRepository = new PhotoRepository();
        BackupUploadLedger ledger = new BackupUploadLedger(context);
        String runKey = "run_" + uid + "_" + getId();
        android.content.SharedPreferences runPrefs = context.getSharedPreferences("photovault_backup_run_counts", Context.MODE_PRIVATE);
        int success = runPrefs.getInt(runKey, 0), failures = 0;
        int imageSuccess = 0;
        int videoSuccess = 0;
        boolean retryableFailure = false;
        String lastError = "";

        try {
            for (int index = 0; index < pending.size(); index++) {
                if (isStopped() || Thread.currentThread().isInterrupted()) return Result.retry();
                UserInfo activeUser = authRepository.isLoggedIn() ? authRepository.getCurrentUser() : null;
                if (activeUser == null || !uid.equals(activeUser.getId())) {
                    BackupWorkScheduler.cancelPeriodic(context, uid);
                    return Result.success(output(success, failures, pending.size() - index,
                            "Phiên hoặc tài khoản đã thay đổi; đã dừng backup để tránh dùng nhầm tài khoản."));
                }
                if (!prefs.isBackupEnabled() || prefs.isPaused()) {
                    BackupWorkScheduler.cancelPeriodic(context, uid);
                    runPrefs.edit().remove(runKey).apply();
                    return Result.success(output(success, failures, pending.size() - index, "Backup đã tạm dừng."));
                }
                LocalMediaPhoto local = pending.get(index);
                if (!hasMediaReadPermission(context, local.isVideo())) {
                    if (success > 0) prefs.setLastSyncTime(System.currentTimeMillis());
                    runPrefs.edit().remove(runKey).apply();
                    return Result.success(output(success, failures, pending.size() - index,
                            "Quyền đọc ảnh đã bị thu hồi; các ảnh còn lại được giữ để thử lại sau."));
                }
                if (SystemClock.elapsedRealtime() - runStartedAt >= MAX_RUN_MILLIS) return Result.retry();
                boolean hasUploadedObject = LocalMediaPhoto.STATUS_METADATA_PENDING.equals(local.getBackupStatus())
                        && local.getCloudUrl() != null && !local.getCloudUrl().isEmpty();
                String url;
                String storagePath;
                if (hasUploadedObject) {
                    url = local.getCloudUrl();
                    storagePath = storageRepository.objectPathFromPublicUrl(url);
                    if (storagePath.isEmpty()) storagePath = stableStoragePath(uid, local);
                } else {
                    AtomicReference<String> uploadedUrl = new AtomicReference<>();
                    AtomicReference<String> uploadError = new AtomicReference<>();
                    CountDownLatch uploadLatch = new CountDownLatch(1);
                    String fileName = stableStoragePath(uid, local);
                    setUploadingProgress(index, pending.size(), success, failures, local, imageSuccess, videoSuccess);
                    storageRepository.uploadMediaAsync(context, Uri.parse(local.getContentUri()), fileName,
                            resolveMimeType(local), new StorageRepository.UploadCallback() {
                        @Override public void onSuccess(String publicUrl) {
                            try (BackupUploadLedger callbackLedger = new BackupUploadLedger(context)) {
                                callbackLedger.recordUpload(uid, local.getContentUri(), local.getDateModified(), publicUrl);
                            } catch (RuntimeException ignored) { }
                            uploadedUrl.set(publicUrl);
                            uploadLatch.countDown();
                        }
                        @Override public void onError(String error) {
                            uploadError.set(error);
                            uploadLatch.countDown();
                        }
                    });
                    if (!await(uploadLatch, CALLBACK_TIMEOUT_SECONDS)) {
                        failures++; retryableFailure = true; lastError = "Hết thời gian upload: " + local.getDisplayName();
                        setProgress(index + 1, pending.size(), success, failures, imageSuccess, videoSuccess);
                        continue;
                    }
                    url = uploadedUrl.get();
                    storagePath = fileName;
                    if (url == null) {
                        failures++;
                        retryableFailure = true;
                        lastError = uploadError.get() == null ? "Supabase Storage không trả về URL ảnh." : uploadError.get();
                        setProgress(index + 1, pending.size(), success, failures, imageSuccess, videoSuccess);
                        continue;
                    }
                    try {
                        ledger.recordUpload(uid, local.getContentUri(), local.getDateModified(), url);
                    } catch (RuntimeException ignored) { }
                }

                setUploadingProgress(index, pending.size(), success, failures, local, imageSuccess, videoSuccess);
                Photo photo = new Photo(
                        local.getCloudPhotoId() == null ? "" : local.getCloudPhotoId(),
                        uid,
                        local.getDisplayName() != null ? local.getDisplayName() : "Photo_" + local.getMediaId(),
                        url,
                        url,
                        System.currentTimeMillis(),
                        local.isVideo() ? "video" : "image",
                        false,
                        false,
                        0L,
                        local.getMediaId(),
                        local.getContentUri(),
                        local.getDateModified(),
                        java.util.Collections.emptyList(),
                        "supabase",
                        computeContentHash(context, Uri.parse(local.getContentUri())),
                        storagePath
                );
                if (!prefs.isBackupEnabled() || prefs.isPaused()) {
                    // The Storage object, if already uploaded, is recoverable from the
                    // UID+URI ledger. Leave its metadata pending for a later enabled run.
                    runPrefs.edit().remove(runKey).apply();
                    if (success > 0) prefs.setLastSyncTime(System.currentTimeMillis());
                    return Result.success(output(success, failures, pending.size() - index,
                            "Backup Ä‘Ã£ táº¡m dá»«ng; media hiá»‡n táº¡i sáº½ Ä‘Æ°á»£c tiáº¿p tá»¥c sau."));
                }
                activeUser = authRepository.isLoggedIn() ? authRepository.getCurrentUser() : null;
                if (activeUser == null || !uid.equals(activeUser.getId())) {
                    // The Storage object is retained and the ledger can recover metadata on a later login.
                    return Result.success(output(success, failures + 1, pending.size() - index,
                            "Tài khoản đổi giữa lúc tải; metadata được giữ để thử lại sau."));
                }
                CountDownLatch metadataLatch = new CountDownLatch(1);
                AtomicBoolean metadataSaved = new AtomicBoolean(false);
                AtomicReference<String> metadataError = new AtomicReference<>();
                photoRepository.saveBackupPhotoAsync(photo, new PhotoRepository.Callback<Void>() {
                    @Override public void onSuccess(Void result) { metadataSaved.set(true); metadataLatch.countDown(); }
                    @Override public void onError(String error) { metadataError.set(error); metadataLatch.countDown(); }
                });
                if (!await(metadataLatch, 45)) {
                    failures++; retryableFailure = true; lastError = "Hết thời gian lưu metadata: " + local.getDisplayName();
                } else if (metadataSaved.get()) {
                    success++;
                    runPrefs.edit().putInt(runKey, success).apply();
                    if (local.isVideo()) videoSuccess++; else imageSuccess++;
                    try { ledger.remove(uid, local.getContentUri()); } catch (RuntimeException ignored) { }
                } else {
                    failures++; retryableFailure = true;
                    lastError = metadataError.get() == null ? "Không lưu được metadata." : metadataError.get();
                }
                setProgress(index + 1, pending.size(), success, failures, imageSuccess, videoSuccess);
            }
        } finally {
            ledger.close();
        }

        if (!prefs.isBackupEnabled() || prefs.isPaused()) {
            BackupWorkScheduler.cancelPeriodic(context, uid);
            runPrefs.edit().remove(runKey).apply();
            if (success > 0) prefs.setLastSyncTime(System.currentTimeMillis());
            return Result.success(output(success, failures, Math.max(0, pending.size() - success), "Backup đã tạm dừng."));
        }
        if (success > 0) prefs.setLastSyncTime(System.currentTimeMillis());
        if (retryableFailure) return Result.retry();

        AtomicReference<PhotoBackupManager.BackupStats> finalStats = new AtomicReference<>();
        CountDownLatch finalScanLatch = new CountDownLatch(1);
        new PhotoBackupManager().processBackupState(context, uid, new PhotoBackupManager.ManagerCallback() {
            @Override public void onStatsReady(PhotoBackupManager.BackupStats stats, List<LocalMediaPhoto> all) {
                finalStats.set(stats); finalScanLatch.countDown();
            }
            @Override public void onError(String error) { finalScanLatch.countDown(); }
        });
        int remaining = Math.max(0, pending.size() - success - failures);
        if (await(finalScanLatch, 45) && finalStats.get() != null) remaining = finalStats.get().remainingCount;
        failures = remaining;
        Data resultData = output(success, failures, remaining, lastError,
                (initialStats == null ? 0 : initialStats.backedUpImages) + imageSuccess,
                (initialStats == null ? 0 : initialStats.backedUpVideos) + videoSuccess);
        runPrefs.edit().remove(runKey).apply();
        return Result.success(resultData);
    }

    private boolean await(CountDownLatch latch, long timeoutSeconds) {
        try { return latch.await(timeoutSeconds, TimeUnit.SECONDS); }
        catch (InterruptedException e) { Thread.currentThread().interrupt(); return false; }
    }

    private String computeContentHash(Context context, Uri uri) {
        try (java.io.InputStream input = context.getContentResolver().openInputStream(uri)) {
            if (input == null) return null;
            java.security.MessageDigest digest = java.security.MessageDigest.getInstance("SHA-256");
            byte[] buffer = new byte[32 * 1024];
            int read;
            while ((read = input.read(buffer)) != -1) digest.update(buffer, 0, read);
            StringBuilder result = new StringBuilder(64);
            for (byte value : digest.digest()) result.append(String.format(java.util.Locale.US, "%02x", value & 0xff));
            return result.toString();
        } catch (Exception ignored) {
            return null;
        }
    }

    private String stableStoragePath(String uid, LocalMediaPhoto local) {
        String identity = local.getContentUri() + "|" + local.getDateModified() + "|" + (local.isVideo() ? "video" : "image");
        String hash;
        try {
            byte[] digest = java.security.MessageDigest.getInstance("SHA-256").digest(identity.getBytes(java.nio.charset.StandardCharsets.UTF_8));
            StringBuilder out = new StringBuilder(64);
            for (byte value : digest) out.append(String.format(java.util.Locale.US, "%02x", value & 0xff));
            hash = out.toString();
        } catch (Exception e) {
            hash = java.util.UUID.nameUUIDFromBytes(identity.getBytes(java.nio.charset.StandardCharsets.UTF_8)).toString().replace("-", "");
        }
        String mime = resolveMimeType(local);
        String extension = "bin";
        if (mime != null) {
            if (mime.equalsIgnoreCase("image/jpeg")) extension = "jpg";
            else if (mime.equalsIgnoreCase("image/png")) extension = "png";
            else if (mime.equalsIgnoreCase("image/webp")) extension = "webp";
            else if (mime.equalsIgnoreCase("image/gif")) extension = "gif";
            else if (mime.equalsIgnoreCase("video/mp4")) extension = "mp4";
            else if (mime.equalsIgnoreCase("video/quicktime")) extension = "mov";
            else if (mime.equalsIgnoreCase("video/webm")) extension = "webm";
            else if (mime.equalsIgnoreCase("video/3gpp")) extension = "3gp";
        }
        if (extension.equals("bin") && local.getDisplayName() != null) {
            int dot = local.getDisplayName().lastIndexOf('.');
            if (dot >= 0 && dot + 1 < local.getDisplayName().length()) {
                String candidate = local.getDisplayName().substring(dot + 1).toLowerCase(java.util.Locale.US);
                if (candidate.matches("[a-z0-9]{1,8}")) extension = candidate;
            }
        }
        return uid + "/" + (local.isVideo() ? "videos/" : "photos/") + hash + "." + extension;
    }

    private String resolveMimeType(LocalMediaPhoto local) {
        String mime = local.getMimeType();
        if (mime != null && mime.contains("/")) return mime;
        String name = local.getDisplayName();
        if (name != null) {
            int dot = name.lastIndexOf('.');
            if (dot >= 0 && dot + 1 < name.length()) {
                String inferred = android.webkit.MimeTypeMap.getSingleton()
                        .getMimeTypeFromExtension(name.substring(dot + 1).toLowerCase(java.util.Locale.US));
                if (inferred != null) return inferred;
            }
        }
        return "application/octet-stream";
    }

    private boolean hasImageReadPermission(Context context) {
        if (Build.VERSION.SDK_INT >= 34) {
            return ContextCompat.checkSelfPermission(context, Manifest.permission.READ_MEDIA_IMAGES) == PackageManager.PERMISSION_GRANTED
                    || ContextCompat.checkSelfPermission(context, Manifest.permission.READ_MEDIA_VISUAL_USER_SELECTED) == PackageManager.PERMISSION_GRANTED;
        }
        String permission = Build.VERSION.SDK_INT >= 33
                ? Manifest.permission.READ_MEDIA_IMAGES : Manifest.permission.READ_EXTERNAL_STORAGE;
        return ContextCompat.checkSelfPermission(context, permission) == PackageManager.PERMISSION_GRANTED;
    }

    private boolean hasMediaReadPermission(Context context, boolean video) {
        if (Build.VERSION.SDK_INT >= 34 && ContextCompat.checkSelfPermission(context,
                Manifest.permission.READ_MEDIA_VISUAL_USER_SELECTED) == PackageManager.PERMISSION_GRANTED) return true;
        if (Build.VERSION.SDK_INT >= 33) {
            String permission = video ? Manifest.permission.READ_MEDIA_VIDEO : Manifest.permission.READ_MEDIA_IMAGES;
            return ContextCompat.checkSelfPermission(context, permission) == PackageManager.PERMISSION_GRANTED;
        }
        return ContextCompat.checkSelfPermission(context, Manifest.permission.READ_EXTERNAL_STORAGE) == PackageManager.PERMISSION_GRANTED;
    }

    private boolean hasAnyMediaReadPermission(Context context) {
        if (Build.VERSION.SDK_INT >= 34
                && ContextCompat.checkSelfPermission(context, Manifest.permission.READ_MEDIA_VISUAL_USER_SELECTED) == PackageManager.PERMISSION_GRANTED) return true;
        if (Build.VERSION.SDK_INT >= 33) {
            return ContextCompat.checkSelfPermission(context, Manifest.permission.READ_MEDIA_IMAGES) == PackageManager.PERMISSION_GRANTED
                    || ContextCompat.checkSelfPermission(context, Manifest.permission.READ_MEDIA_VIDEO) == PackageManager.PERMISSION_GRANTED;
        }
        return ContextCompat.checkSelfPermission(context, Manifest.permission.READ_EXTERNAL_STORAGE) == PackageManager.PERMISSION_GRANTED;
    }

                .putInt("success", success).putInt("failures", failures).putInt("uploading", 0)
                .putInt("uploaded_images", images).putInt("uploaded_videos", videos).build());
    }

                                     int images, int videos) {

                .putInt("success", success).putInt("failures", failures).putInt("uploading", 1)
                .putInt("uploaded_images", images).putInt("uploaded_videos", videos)
                .putString("current_name", media.getDisplayName()).putString("current_type", media.isVideo() ? "video" : "image").build());
    }

                .putInt("remaining", remaining).putString("error", error).build();
    }

        return new Data.Builder().putInt("success", success).putInt("failures", failures)
                .putInt("remaining", remaining).putString("error", error)
                .putInt("uploaded_images", images).putInt("uploaded_videos", videos).build();
    }

package com.example.th6.services;

import android.content.ContentResolver;
import android.content.ContentUris;
import android.content.Context;
import android.database.Cursor;
import android.net.Uri;
import android.os.Build;
import android.os.Handler;
import android.os.Looper;
import android.provider.MediaStore;
import android.util.Log;

import com.example.th6.models.LocalMediaPhoto;

import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.List;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

public class DevicePhotoScanner {
    private static final String TAG = "DevicePhotoScanner";
    private static final ExecutorService SCAN_EXECUTOR = Executors.newSingleThreadExecutor();
    private final Handler mainHandler = new Handler(Looper.getMainLooper());

    public interface ScanCallback {
        void onScanComplete(List<LocalMediaPhoto> localPhotos);
        void onScanError(String error);
    }

    public void scanPhotos(Context context, ScanCallback callback) {
        scan(context, callback, false);
    }

    /** Scans only images and reports revoked image permission instead of returning an empty queue. */
    public void scanImages(Context context, ScanCallback callback) {
        scan(context, callback, true);
    }

    private void scan(Context context, ScanCallback callback, boolean imagesOnly) {
        SCAN_EXECUTOR.execute(() -> {
            try {
                ContentResolver resolver = context.getContentResolver();
                List<LocalMediaPhoto> media = new ArrayList<>();
                scanCollection(resolver, getExternalCollectionUri(false), false, media, imagesOnly);
                if (!imagesOnly) scanCollection(resolver, getExternalCollectionUri(true), true, media, false);
                Collections.sort(media, (a, b) -> Long.compare(b.getTimelineTimestamp(), a.getTimelineTimestamp()));
                mainHandler.post(() -> {
                    if (callback != null) callback.onScanComplete(media);
                });
            } catch (Exception e) {
                mainHandler.post(() -> {
                    if (callback != null) callback.onScanError("Media scan failed: " + e.getLocalizedMessage());
                });
            }
        });
    }

    private void scanCollection(ContentResolver resolver, Uri collection, boolean video,
                                List<LocalMediaPhoto> result, boolean failOnSecurityError) {
        String[] projection = video
                ? new String[]{MediaStore.Video.Media._ID, MediaStore.Video.Media.DISPLAY_NAME,
                    MediaStore.Video.Media.MIME_TYPE, MediaStore.Video.Media.SIZE,
                    MediaStore.Video.Media.WIDTH, MediaStore.Video.Media.HEIGHT,
                    MediaStore.Video.Media.DATE_TAKEN, MediaStore.Video.Media.DATE_ADDED,
                    MediaStore.Video.Media.DATE_MODIFIED, MediaStore.Video.Media.DURATION}
                : new String[]{MediaStore.Images.Media._ID, MediaStore.Images.Media.DISPLAY_NAME,
                    MediaStore.Images.Media.MIME_TYPE, MediaStore.Images.Media.SIZE,
                    MediaStore.Images.Media.WIDTH, MediaStore.Images.Media.HEIGHT,
                    MediaStore.Images.Media.DATE_TAKEN, MediaStore.Images.Media.DATE_ADDED,
                    MediaStore.Images.Media.DATE_MODIFIED};
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            projection = Arrays.copyOf(projection, projection.length + 1);
            projection[projection.length - 1] = MediaStore.MediaColumns.IS_PENDING;
        }
        int itemCount = 0;
        try (Cursor cursor = resolver.query(collection, projection, null, null, null)) {
            if (cursor == null) {
                Log.i(TAG, (video ? "Video" : "Image") + " query returned a null cursor");
                if (failOnSecurityError) throw new IllegalStateException("MediaStore did not return an image cursor.");
                return;
            }
            int idCol = cursor.getColumnIndexOrThrow(MediaStore.MediaColumns._ID);
            int nameCol = cursor.getColumnIndexOrThrow(MediaStore.MediaColumns.DISPLAY_NAME);
            int mimeCol = cursor.getColumnIndexOrThrow(MediaStore.MediaColumns.MIME_TYPE);
            int sizeCol = cursor.getColumnIndexOrThrow(MediaStore.MediaColumns.SIZE);
            int widthCol = cursor.getColumnIndexOrThrow(MediaStore.MediaColumns.WIDTH);
            int heightCol = cursor.getColumnIndexOrThrow(MediaStore.MediaColumns.HEIGHT);
            int takenCol = cursor.getColumnIndex(MediaStore.Images.Media.DATE_TAKEN);
            int addedCol = cursor.getColumnIndexOrThrow(MediaStore.MediaColumns.DATE_ADDED);
            int modifiedCol = cursor.getColumnIndexOrThrow(MediaStore.MediaColumns.DATE_MODIFIED);
            int pendingCol = Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q
                    ? cursor.getColumnIndex(MediaStore.MediaColumns.IS_PENDING) : -1;
            int durationCol = video ? cursor.getColumnIndexOrThrow(MediaStore.Video.Media.DURATION) : -1;

            while (cursor.moveToNext()) {
                if (pendingCol >= 0 && cursor.getInt(pendingCol) != 0) continue;
                long id = cursor.getLong(idCol);
                long timestamp = getModifiedTimestamp(cursor, modifiedCol);
                Uri contentUri = ContentUris.withAppendedId(collection, id);
                LocalMediaPhoto item = new LocalMediaPhoto(id, contentUri.toString(),
                        cursor.getString(nameCol), cursor.getString(mimeCol), cursor.getLong(sizeCol),
                        cursor.getInt(widthCol), cursor.getInt(heightCol), timestamp);
                item.setDateTaken(getCaptureTimestamp(cursor, takenCol, addedCol, modifiedCol));
                item.setVideo(video);
                if (video && durationCol >= 0) item.setDurationMs(Math.max(0, cursor.getLong(durationCol)));
                result.add(item);
                itemCount++;
            }
            Log.i(TAG, (video ? "Video" : "Image") + " query returned " + itemCount + " items");
        } catch (SecurityException e) {
            Log.w(TAG, (video ? "Video" : "Image") + " query denied by media permissions", e);
            if (failOnSecurityError) throw e;
        } catch (RuntimeException e) {
            Log.e(TAG, (video ? "Video" : "Image") + " MediaStore query failed", e);
            if (failOnSecurityError) throw e;
        }
    }

    private Uri getExternalCollectionUri(boolean video) {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            return video
                    ? MediaStore.Video.Media.getContentUri(MediaStore.VOLUME_EXTERNAL)
                    : MediaStore.Images.Media.getContentUri(MediaStore.VOLUME_EXTERNAL);
        }
        return video ? MediaStore.Video.Media.EXTERNAL_CONTENT_URI : MediaStore.Images.Media.EXTERNAL_CONTENT_URI;
    }

    private long getModifiedTimestamp(Cursor cursor, int modifiedCol) {
        long modified = cursor.getLong(modifiedCol);
        // This value is persisted as local_date_modified and used for change detection.
        // DATE_TAKEN/DATE_ADDED describe capture/import time and are not safe revision signals.
        return modified > 0 ? modified * 1000L : 0L;
    }

    private long getCaptureTimestamp(Cursor cursor, int takenCol, int addedCol, int modifiedCol) {
        long taken = takenCol >= 0 ? cursor.getLong(takenCol) : 0L;
        if (taken > 0) return taken;
        long added = cursor.getLong(addedCol);
        if (added > 0) return added * 1000L;
        long modified = cursor.getLong(modifiedCol);
        return modified > 0 ? modified * 1000L : 0L;
    }
}
